"""
Security-Modul für ELION Dashboard
Implementiert Token-Validierung, Rate-Limiting und Sicherheitsprotokollierung.
"""

import logging, time, secrets
from pathlib import Path
from typing import Dict
from fastapi import HTTPException
from fastapi.security import HTTPAuthorizationCredentials

# Logging
log_dir = Path("logs"); log_dir.mkdir(exist_ok=True)
formatter = logging.Formatter("%(asctime)s - %(levelname)s - %(message)s")
logger = logging.getLogger("security"); logger.setLevel(logging.INFO)
fh = logging.FileHandler(log_dir / "security.log"); fh.setFormatter(formatter)
logger.addHandler(fh)

# ----------------------------------------------------------
# RateLimiter
# ----------------------------------------------------------
class RateLimiter:
    def __init__(self, requests_per_minute: int = 60):
        self.rate_limit = requests_per_minute
        self.window = 60
        self._reqs: Dict[str, list] = {}

    def _clean(self, t: str):
        now = time.time()
        if t in self._reqs:
            self._reqs[t] = [ts for ts in self._reqs[t] if now - ts < self.window]

    def is_allowed(self, t: str) -> bool:
        self._clean(t); now = time.time()
        if t not in self._reqs: self._reqs[t] = []
        if len(self._reqs[t]) >= self.rate_limit: return False
        self._reqs[t].append(now); return True

    def limit(self):
        from functools import wraps
        from fastapi.security import HTTPBearer
        security = HTTPBearer()

        def decorator(func):
            @wraps(func)
            async def wrapper(*args, **kwargs):
                token = kwargs.get("token")
                if not token:
                    for a in args:
                        if isinstance(a, HTTPAuthorizationCredentials):
                            token = a.credentials; break
                if not token:
                    raise HTTPException(status_code=401, detail="Token erforderlich")
                if not self.is_allowed(token):
                    raise HTTPException(status_code=429, detail="Rate limit überschritten")
                return await func(*args, **kwargs)
            return wrapper
        return decorator

# ----------------------------------------------------------
# Token-Handling
# ----------------------------------------------------------
def _read_env_token() -> str:
    env_path = Path(".env")
    if env_path.exists():
        return env_path.read_text(encoding="utf-8").strip()
    new_token = secrets.token_urlsafe(32)
    env_path.write_text(new_token, encoding="utf-8")
    logger.info(f"Neuer Token generiert und in .env gespeichert.")
    return new_token

def verify_token(token: str) -> bool:
    try:
        valid = _read_env_token().strip()
        result = token.strip() == valid
        logger.info(f"Tokenprüfung ok={result}")
        return result
    except Exception as e:
        logger.error(f"Fehler bei Token-Validierung: {e}")
        return False

"""
Security-Modul für ELION Dashboard
Implementiert Token-Validierung, Rate-Limiting und Sicherheitsprotokollierung.
"""

import logging, time, secrets
from pathlib import Path
from typing import Dict
from fastapi import HTTPException
from fastapi.security import HTTPAuthorizationCredentials

# ----------------------------------------------------------
# Logging
# ----------------------------------------------------------
log_dir = Path("logs"); log_dir.mkdir(exist_ok=True)
formatter = logging.Formatter("%(asctime)s - %(levelname)s - %(message)s")
logger = logging.getLogger("security"); logger.setLevel(logging.INFO)
fh = logging.FileHandler(log_dir / "security.log"); fh.setFormatter(formatter)
logger.addHandler(fh)


# ----------------------------------------------------------
# RateLimiter
# ----------------------------------------------------------
class RateLimiter:
    """Begrenzt Requests pro Token"""
    def __init__(self, requests_per_minute: int = 60):
        self.rate_limit = requests_per_minute
        self.window = 60  # Sekunden
        self._reqs: Dict[str, list] = {}

    def _clean(self, key: str):
        now = time.time()
        if key in self._reqs:
            self._reqs[key] = [ts for ts in self._reqs[key] if now - ts < self.window]

    def is_allowed(self, key: str) -> bool:
        self._clean(key)
        now = time.time()
        if key not in self._reqs:
            self._reqs[key] = []
        if len(self._reqs[key]) >= self.rate_limit:
            return False
        self._reqs[key].append(now)
        return True

    def limit(self):
        """Decorator für Endpunkte"""
        from functools import wraps
        from fastapi.security import HTTPBearer
        security = HTTPBearer()

        def decorator(func):
            @wraps(func)
            async def wrapper(*args, **kwargs):
                creds = kwargs.get("token")
                token_value = None

                # Extrahiere Token-Zeichenkette aus Argumenten
                if isinstance(creds, HTTPAuthorizationCredentials):
                    token_value = creds.credentials
                elif isinstance(creds, str):
                    token_value = creds
                else:
                    # Prüfe auch Positional Arguments
                    for a in args:
                        if isinstance(a, HTTPAuthorizationCredentials):
                            token_value = a.credentials
                            break

                if not token_value:
                    raise HTTPException(status_code=401, detail="Token erforderlich")

                if not self.is_allowed(token_value):
                    raise HTTPException(status_code=429, detail="Rate limit überschritten")

                return await func(*args, **kwargs)
            return wrapper
        return decorator


# ----------------------------------------------------------
# Token
# ----------------------------------------------------------
def _read_env_token() -> str:
    env_path = Path(".env")
    if env_path.exists():
        return env_path.read_text(encoding="utf-8").strip()
    new_token = secrets.token_urlsafe(32)
    env_path.write_text(new_token, encoding="utf-8")
    logger.info("Neuer Token generiert und in .env gespeichert.")
    return new_token


def verify_token(token: str) -> bool:
    """Vergleicht eingehenden Token mit .env"""
    try:
        valid = _read_env_token().strip()
        ok = token.strip() == valid
        logger.info(f"Tokenprüfung ok={ok}")
        return ok
    except Exception as e:
        logger.error(f"Fehler bei Token-Validierung: {e}")
        return False
