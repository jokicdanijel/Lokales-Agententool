#!/usr/bin/env bash
set -euo pipefail
ROOT="$(cd "$(dirname "$0")" && pwd)"
cd "$ROOT"

# 1) Archivprüfung (heute)
D=$(date +%Y/%m/%d); AP="$ROOT/archivp/$D"
echo "[Archiv] $AP"
test -d "$AP" || { echo "FEHLT: Tagesordner"; exit 1; }
CMDc=$(ls "$AP"/*_CMD.json 2>/dev/null | wc -l || true)
RESPc=$(ls "$AP"/*_RESP.json 2>/dev/null | wc -l || true)
test "$CMDc" -ge 1 && test "$RESPc" -ge 1 || { echo "FEHLT: CMD/RESP Safepoints"; exit 1; }
grep -q '"strict"[[:space:]]*:[[:space:]]*true' "$AP"/*.json || { echo "FEHLT: strict:true"; exit 1; }

# 2) GitHub-Validierung
if git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
  git status --porcelain=v1 | tee /tmp/git_dirty.txt
  test ! -s /tmp/git_dirty.txt || { echo "GIT DIRTY"; exit 1; }
  git log -1 --oneline >/dev/null || { echo "KEIN LETZTER COMMIT"; exit 1; }
else
  echo "KEIN GIT REPO"; exit 1
fi

# 3) MAC_DIR_SYSTEM.json aktualisieren + prüfen
python3 generate_macdir.py
jq '.system' MAC_DIR_SYSTEM.json >/dev/null

echo "PREFLIGHT OK"
SH

Code kopieren
#!/usr/bin/env bash
set -euo pipefail

# --- Konfig ---
PROJECT_ROOT="${PROJECT_ROOT:-/home/mint/projekte/open-webui}"
ARCHIV_ROOT="${ARCHIV_ROOT:-/archivp}"
TODAY_Y=$(date +%Y) ; TODAY_M=$(date +%m) ; TODAY_D=$(date +%d)
TODAY_DIR="$ARCHIV_ROOT/$TODAY_Y/$TODAY_M/$TODAY_D"

# env.sh nachladen, wenn vorhanden
if [[ -f "$PROJECT_ROOT/env.sh" ]]; then
  # shellcheck disable=SC1091
  source "$PROJECT_ROOT/env.sh"
fi

usage() {
  cat <<USAGE
Preflight für Portier/ELION
  --archive-check   Archiv prüfen (Safepoints + "strict": true)
  --git-check       Git-Status + letzter Commit prüfen
  --macdir          MAC_DIR_SYSTEM.json generieren/aktualisieren
  --all             alles ausführen
USAGE
}

log() { printf "\n\033[1m>> %s\033[0m\n" "$*"; }
warn(){ printf "\033[33m[WARN]\033[0m %s\n" "$*"; }
ok()  { printf "\033[32m[OK]\033[0m %s\n" "$*"; }
err() { printf "\033[31m[ERR]\033[0m %s\n" "$*"; }

archive_check() {
  log "Archivprüfung: $TODAY_DIR"
  if [[ ! -d "$TODAY_DIR" ]]; then
    warn "Heutiger Archivordner existiert nicht: $TODAY_DIR"
    return 1
  fi

  shopt -s nullglob
  mapfile -t CMD_FILES < <(ls "$TODAY_DIR"/*CMD.json 2>/dev/null || true)
  mapfile -t RESP_FILES < <(ls "$TODAY_DIR"/*RESP.json 2>/dev/null || true)
  shopt -u nullglob

  [[ ${#CMD_FILES[@]} -gt 0 ]] || warn "Kein CMD-Safepoint gefunden."
  [[ ${#RESP_FILES[@]} -gt 0 ]] || warn "Kein RESP-Safepoint gefunden."

  if command -v jq >/dev/null 2>&1; then
    BAD=0
    for f in "${CMD_FILES[@]}" "${RESP_FILES[@]}"; do
      [[ -f "$f" ]] || continue
      if ! jq -e '.strict == true' "$f" >/dev/null 2>&1; then
        warn "\"strict\": true fehlt in: $f"
        BAD=1
      fi
    done
    [[ $BAD -eq 0 ]] && ok "Alle geprüften Safepoints mit \"strict\": true."
  else
    warn "jq nicht installiert – Inhalt nicht validiert (nur Anwesenheit geprüft)."
  fi

  # optionale Index-Datei
  if [[ -f "$ARCHIV_ROOT/index.jsonl" ]]; then
    ok "index.jsonl vorhanden."
  else
    warn "index.jsonl nicht gefunden (optional)."
  fi
}

git_check() {
  log "GitHub-Validierung im Projekt: $PROJECT_ROOT"
  if [[ ! -d "$PROJECT_ROOT/.git" ]]; then
    err "Kein Git-Repository unter $PROJECT_ROOT"
    return 1
  fi
  ( cd "$PROJECT_ROOT"
    git status
    echo
    git log -1 --oneline --decorate --date=local --pretty=format:'%C(yellow)%h %Creset%s %Cgreen(%cr) %C(bold blue)<%an>%Creset'
  )
  ok "Git geprüft."
}

macdir_update() {
  log "MAC_DIR_SYSTEM.json aktualisieren"
  cd "$PROJECT_ROOT"

  if [[ -x ./start_connector.sh ]]; then
    bash ./start_connector.sh --macdir || warn "start_connector.sh --macdir fehlgeschlagen."
  elif [[ -f ./generate_macdir.py ]]; then
    python3 ./generate_macdir.py || warn "generate_macdir.py fehlgeschlagen."
  else
    warn "Weder start_connector.sh noch generate_macdir.py gefunden."
    return 1
  fi

  if [[ -f MAC_DIR_SYSTEM.json ]]; then
    if command -v jq >/dev/null 2>&1; then
      echo "System-Feld:"
      jq .system MAC_DIR_SYSTEM.json || true
    else
      cat MAC_DIR_SYSTEM.json
    fi
    ok "MAC_DIR_SYSTEM.json aktualisiert."
  else
    err "MAC_DIR_SYSTEM.json wurde nicht erzeugt."
    return 1
  fi
}

run_all() {
  archive_check || true
  git_check || true
  macdir_update || true
}

main() {
  [[ $# -eq 0 ]] && { usage; exit 1; }

  case "${1:-}" in
    --archive-check) archive_check ;;
    --git-check)     git_check ;;
    --macdir)        macdir_update ;;
    --all)           run_all ;;
    -h|--help)       usage ;;
    *)               usage; exit 2 ;;
  esac
}

main "$@"

#!/usr/bin/env bash
set -euo pipefail

TS="$(date -u +%Y%m%d_%H%M%S)"
PROJECT_ROOT="/srv/projects/ai-assistant-jarvis"
SRC_LIST=(
  "${PROJECT_ROOT}/app"
  "${PROJECT_ROOT}/scripts"
  "${PROJECT_ROOT}/Dockerfile"
  "${PROJECT_ROOT}/docker-compose.yml"
  "${PROJECT_ROOT}/requirements.txt"
  "${PROJECT_ROOT}/PROJECT_SPEC.yaml"
  "${PROJECT_ROOT}/.env.example"
)
ARCH_ROOT="/backup1/aaj-snapshots"
SPARE_ROOT="/backup2/aaj-snapshots"   # optional
LOG_DIR="/srv/logs/aaj/archivator"
OUT_DIR="${ARCH_ROOT}/${TS}"
MANIFEST="${OUT_DIR}/MANIFEST.json"
TAR="${OUT_DIR}/aaj_snapshot_${TS}.tar.zst"

mkdir -p "${OUT_DIR}" "${LOG_DIR}"
echo "[INFO] ${TS} Archivator startet" >> "${LOG_DIR}/archivator.log"

# Manifest
{
  echo '{'
  echo '  "timestamp_utc": "'"$(date -u +%Y-%m-%dT%H:%M:%SZ)"'",'
  echo '  "project_root": "'"${PROJECT_ROOT}"'",'
  echo '  "items": ['
  first=1
  for p in "${SRC_LIST[@]}"; do
    if [[ -e "$p" ]]; then
      [[ $first -eq 0 ]] && echo ','
      size=$(du -sb "$p" | awk '{print $1}')
      printf '    {"path": "%s", "size": %s}' "$p" "$size"
      first=0
    fi
  done
  echo
  echo '  ]'
  echo '}'
} > "${MANIFEST}"

# Paket bauen (relativ, ohne absolute Pfade im Inhalt)
( cd "$(dirname "${PROJECT_ROOT}")"
  BASE="$(basename "${PROJECT_ROOT}")"
  tar -I 'zstd -19' -cf "${TAR}" "${BASE}"
)

# Hash erzeugen
sha256sum "${TAR}" | tee "${TAR}.sha256" > /dev/null

# Rotation: behalte 7 daily, 4 weekly, 3 monthly
find "${ARCH_ROOT}" -maxdepth 1 -type d -name '20????????_??????' -printf '%T@ %p\n' \
  | sort -n | awk 'NR<=100{print $2}' | while read -r d; do
    base=$(basename "$d")
    y=${base:0:4}; m=${base:4:2}; w=$(date -ud "${y}-${m}-01" +%U)
    # primitive Buckets: last 7 days, last 4 weeks, last 3 months
  done > /dev/null 2>&1
# (Für echte Policy könntest du eine separate rotate-Datei nutzen; hier: manuell löschen nach Bedarf)

# Spiegel nach backup2 (optional)
if [[ -d "${SPARE_ROOT}" ]]; then
  mkdir -p "${SPARE_ROOT}"
  rsync -a --delete "${ARCH_ROOT}/" "${SPARE_ROOT}/" >> "${LOG_DIR}/archivator.log" 2>&1 || true
fi

echo "[OK] ${TS} Archiv fertig: ${TAR}" >> "${LOG_DIR}/archivator.log"
echo "${TAR}"
chmod +x /srv/projects/ai-assistant-jarvis/scripts/archivator.sh

  "${PROJECT_ROOT}/app"
  "${PROJECT_ROOT}/scripts"
  "${PROJECT_ROOT}/Dockerfile"
  "${PROJECT_ROOT}/docker-compose.yml"
  "${PROJECT_ROOT}/requirements.txt"
  "${PROJECT_ROOT}/PROJECT_SPEC.yaml"
  "${PROJECT_ROOT}/.env.example"
)
ARCH_ROOT="/backup1/aaj-snapshots"
SPARE_ROOT="/backup2/aaj-snapshots"   # optional
LOG_DIR="/srv/logs/aaj/archivator"
OUT_DIR="${ARCH_ROOT}/${TS}"
MANIFEST="${OUT_DIR}/MANIFEST.json"
TAR="${OUT_DIR}/aaj_snapshot_${TS}.tar.zst"

mkdir -p "${OUT_DIR}" "${LOG_DIR}"
echo "[INFO] ${TS} Archivator startet" >> "${LOG_DIR}/archivator.log"

# Manifest
{
  echo '{'
  echo '  "timestamp_utc": "'"$(date -u +%Y-%m-%dT%H:%M:%SZ)"'",'
  echo '  "project_root": "'"${PROJECT_ROOT}"'",'
  echo '  "items": ['
  first=1
  for p in "${SRC_LIST[@]}"; do
    if [[ -e "$p" ]]; then
      [[ $first -eq 0 ]] && echo ','
      size=$(du -sb "$p" | awk '{print $1}')
      printf '    {"path": "%s", "size": %s}' "$p" "$size"
      first=0
    fi
  done
  echo
  echo '  ]'
  echo '}'
} > "${MANIFEST}"

# Paket bauen (relativ, ohne absolute Pfade im Inhalt)
( cd "$(dirname "${PROJECT_ROOT}")"
  BASE="$(basename "${PROJECT_ROOT}")"
  tar -I 'zstd -19' -cf "${TAR}" "${BASE}"
)

# Hash erzeugen
sha256sum "${TAR}" | tee "${TAR}.sha256" > /dev/null

# Rotation: behalte 7 daily, 4 weekly, 3 monthly
find "${ARCH_ROOT}" -maxdepth 1 -type d -name '20????????_??????' -printf '%T@ %p\n' \
  | sort -n | awk 'NR<=100{print $2}' | while read -r d; do
    base=$(basename "$d")
    y=${base:0:4}; m=${base:4:2}; w=$(date -ud "${y}-${m}-01" +%U)
    # primitive Buckets: last 7 days, last 4 weeks, last 3 months
  done > /dev/null 2>&1
# (Für echte Policy könntest du eine separate rotate-Datei nutzen; hier: manuell löschen nach Bedarf)

# Spiegel nach backup2 (optional)
if [[ -d "${SPARE_ROOT}" ]]; then
  mkdir -p "${SPARE_ROOT}"
  rsync -a --delete "${ARCH_ROOT}/" "${SPARE_ROOT}/" >> "${LOG_DIR}/archivator.log" 2>&1 || true
fi

echo "[OK] ${TS} Archiv fertig: ${TAR}" >> "${LOG_DIR}/archivator.log"
echo "${TAR}"
chmod +x /srv/projects/ai-assistant-jarvis/scripts/archivator.sh
Hinweis: Rotation ist hier bewusst konservativ. Wenn du harte Behaltensregeln willst, füge ich dir archivator_rotate.sh mit Datumseimern hinzu.

10) Archivator einmal testen

ls -lh /backup1/aaj-snapshots/*/
Erwartet: Verzeichnis mit *.tar.zst und *.sha256 + MANIFEST.json.

11) Archivator systemd-Service (System-Modus, damit er auch ohne Login läuft)
Datei: /etc/systemd/system/aaj-archivator.service
[Unit]
Description=AAJ Archivator (Snapshot + Hash + optional Spiegel)
After=network-online.target
Wants=network-online.target

[Service]
Type=oneshot
User=%i
Group=%i
ExecStart=/srv/projects/ai-assistant-jarvis/scripts/archivator.sh
Nice=10
IOSchedulingClass=best-effort
IOSchedulingPriority=7
%i = Instanzbenutzer. Wir starten ihn mit deinem User.

12) Archivator Timer (täglich 02:30)
Datei: /etc/systemd/system/aaj-archivator.timer
[Unit]
Description=AAJ Archivator Timer (daily 02:30)

[Timer]
OnCalendar=*-*-* 02:30:00
Persistent=true
Unit=aaj-archivator@%i.service

[Install]
WantedBy=timers.target

13) Timer aktivieren (als dein User)
sudo systemctl daemon-reload
sudo systemctl enable --now aaj-archivator.timer
# Instanzzuweisung an deinen Benutzer:
sudo systemctl enable --now aaj-archivator@"$USER".service
# Timer-Status:
systemctl list-timers | grep aaj-archivator || true

14) On-Demand Backup (manuell)
sudo systemctl start aaj-archivator@"$USER".service
journalctl -u aaj-archivator@"$USER".service -n 100 -o short-iso --no-pager
ls -lh /backup1/aaj-snapshots/*/

cd /backup1/aaj-snapshots/$(ls /backup1/aaj-snapshots | tail -n1)

Erwartet: OK für die erzeugte Datei.

18) Optional: /srv dauerhaft auf interne SSD (/dev/sdb1)
Zerstörerisch für /dev/sdb – nur wenn leer!
sudo parted -s /dev/sdb mklabel gpt
sudo parted -s /dev/sdb mkpart primary ext4 0% 100%
sudo mkfs.ext4 -L DATA_SRV /dev/sdb1
UUID=$(sudo blkid -s UUID -o value /dev/sdb1)
echo "UUID=${UUID}  /srv  ext4  noatime,defaults  0  2" | sudo tee -a /etc/fstab
sudo mount -a
sudo chown -R "$USER":"$USER" /srv
Dann Modelle/Projekt nach /srv verschieben (falls aktuell unter /), .env bleibt gültig.

19) Optional: HDDs als Backupziele einrichten
# /dev/sda → /backup1
sudo mkfs.ext4 -L BACKUP1 /dev/sda1 2>/dev/null || true
UUID1=$(sudo blkid -s UUID -o value /dev/sda1)
echo "UUID=${UUID1}  /backup1  ext4  noatime,defaults  0  2" | sudo tee -a /etc/fstab

# /dev/sdc → /backup2
sudo mkfs.ext4 -L BACKUP2 /dev/sdc1 2>/dev/null || true
UUID2=$(sudo blkid -s UUID -o value /dev/sdc1)
echo "UUID=${UUID2}  /backup2  ext4  noatime,defaults  0  2" | sudo tee -a /etc/fstab

sudo mount -a
sudo chown -R "$USER":"$USER" /backup1 /backup2

20) Abnahmecheck (Knock-out-Kriterien)
# Koordinator läuft?
systemctl --user is-active aaj-coordinator && echo "Koordinator: OK" || echo "Koordinator: NOK"

# Health OK?
/srv/projects/ai-assistant-jarvis/scripts/healthcheck.sh || true

# Archivator Timer aktiv?
systemctl is-enabled aaj-archivator.timer && echo "Archivator Timer: enabled" || echo "Archivator Timer: disabled"

# Letzter Snapshot vorhanden + Hash OK?
LATEST="$(ls -d /backup1/aaj-snapshots/* 2>/dev/null | tail -n1)"
test -n "$LATEST" && echo "Snapshot: $LATEST" || echo "Snapshot fehlt"
cd "$LATEST" && sha256sum -c *.sha256
Grün, wenn: Koordinator aktiv, Health meldet OK, Timer enabled, Snapshot da & Hash OK.

Was du damit hast
    • Koordinator: Startet/überwacht deine lokale Jarvis-Pipeline (Wakeword+Vosk), loggt sauber.
    • Archivator: Baut versionierte, verifizierbare Snapshots (.tar.zst + sha256 + Manifest), optional gespiegelt.
