```
#!/usr/bin/env bash
set -euo pipefail

# -----------------------------------------------------------------------------
# Portier / ELION Hyper-Dashboard 2.0
# Bootstrap der vier Kernservices (opena1, kordp, archivp, opena2)
# Ports: 12344–12399 erlaubt, 8080 nur für Open Web UI
# -----------------------------------------------------------------------------

ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
cd "$ROOT"

# --- Env laden ---------------------------------------------------------------
if [[ -f ".env" ]]; then
  set -a
  # shellcheck disable=SC1091
  source ".env"
  set +a
fi

# Defaults, falls nicht gesetzt
PORT_WINDOW_START="${PORT_WINDOW_START:-12344}"
PORT_WINDOW_END="${PORT_WINDOW_END:-12399}"
OPENWEBUI_PORT="${OPENWEBUI_PORT:-8080}"

OPENA1_PORT="${OPENA1_PORT:-12344}"
KORDP_PORT="${KORDP_PORT:-12345}"
ARCHIVP_PORT="${ARCHIVP_PORT:-12346}"
OPENA2_PORT="${OPENA2_PORT:-12347}"

BIND_ADDR="${BIND_ADDR:-127.0.0.1}"
LOG_DIR="${LOG_DIR:-$ROOT/logs}"
VENV_DIR="${VENV_DIR:-$ROOT/1.portier_openai/venv313}"
LOCAL_VENV_LINK="$ROOT/.venv"

DIR_OPENA1="${DIR_OPENA1:-3.opena1_coordinator}"
DIR_KORDP="${DIR_KORDP:-5.kordp_scheduler}"
DIR_ARCHIVP="${DIR_ARCHIVP:-1.portier_openai}"
DIR_OPENA2="${DIR_OPENA2:-4.opena2_archivator}"

mkdir -p "$LOG_DIR"

# --- Helpers -----------------------------------------------------------------
port_free() { ! ss -ltn "( sport = :$1 )" | grep -q LISTEN; }
enforce_window() {
  local p="$1"
  if (( p < PORT_WINDOW_START || p > PORT_WINDOW_END )); then
    echo "ERROR: Port $p außerhalb des erlaubten Fensters ${PORT_WINDOW_START}-${PORT_WINDOW_END}" >&2
    exit 1
  fi
}

start_venv() {
  if [[ ! -x "$VENV_DIR/bin/python" ]]; then
    python3.13 -m venv "$VENV_DIR"
    "$VENV_DIR/bin/python" -m pip install --upgrade pip >/dev/null
  fi
  ln -sfn "$VENV_DIR" "$LOCAL_VENV_LINK"
}

detect_cmd() {
  # Ermittelt Startkommando im angegebenen Ordner.
  # Priorität: run.sh > start.sh > uvicorn main:app > uvicorn app:app > python main.py > python app.py
  local d="$1"
  if [[ -x "$d/run.sh" ]]; then echo "bash ./run.sh"; return 0; fi
  if [[ -x "$d/start.sh" ]]; then echo "bash ./start.sh"; return 0; fi
  if [[ -f "$d/main.py" ]] && grep -qE "^app\s*=" "$d/main.py"; then
    echo "$LOCAL_VENV_LINK/bin/uvicorn main:app --host $BIND_ADDR --port"; return 0
  fi
  if [[ -f "$d/app.py" ]] && grep -qE "^app\s*=" "$d/app.py"; then
    echo "$LOCAL_VENV_LINK/bin/uvicorn app:app --host $BIND_ADDR --port"; return 0
  fi
  if [[ -f "$d/main.py" ]]; then echo "$LOCAL_VENV_LINK/bin/python main.py --port"; return 0; fi
  if [[ -f "$d/app.py" ]]; then echo "$LOCAL_VENV_LINK/bin/python app.py --port"; return 0; fi
  # Fallback: module-start falls vorhanden
  if [[ -f "$d/__main__.py" ]]; then echo "$LOCAL_VENV_LINK/bin/python -m $(basename "$d") --port"; return 0; fi
  return 1
}

start_service() {
  local name="$1" dir="$2" port="$3"
  enforce_window "$port"
  if ! port_free "$port"; then
    echo "ERROR: $name Port :$port ist belegt." >&2
    exit 1
  fi
  if [[ ! -d "$dir" ]]; then
    echo "ERROR: Verzeichnis '$dir' für $name fehlt." >&2
    exit 1
  fi
  local cmd
  if ! cmd="$(detect_cmd "$dir")"; then
    echo "ERROR: Kein Startkommando in '$dir' erkannt (run.sh/start.sh/main.py/app.py)." >&2
    exit 1
  fi

  echo "→ Starte $name @ :$port  (dir: $dir)"
  pushd "$dir" >/dev/null
    # Standardisierte Env
    export PORTIER_ROLE="$name"
    export BIND_ADDR
    export PORT="$port"
    export LOG_LEVEL="${LOG_LEVEL:-INFO}"
    export OPENA1_PORT OPENA2_PORT KORDP_PORT ARCHIVP_PORT
    export OPENA1_LOG_EP KORDP_DISPATCH_EP ARCHIVP_STORE_EP OPENA2_FINALIZE_EP

    # Portparameter anhängen falls nötig
    if [[ "$cmd" =~ "--port"$ ]]; then
      cmd="$cmd $port"
    fi

    # Start
    nohup bash -lc "$cmd" \
      > "$LOG_DIR/${name}.log" 2>&1 < /dev/null &
  popd >/dev/null
}

wait_health() {
  local port="$1" name="$2" path="${3:-/health}"
  for i in {1..30}; do
    if curl -fsS "http://127.0.0.1:${port}${path}" >/dev/null; then
      echo "OK ${name} @ :${port}${path}"
      return 0
    fi
    sleep 1
  done
  echo "FAIL ${name} @ :${port}${path}" >&2
  return 1
}

# --- Guards ------------------------------------------------------------------
if ss -ltn "( sport = :${OPENWEBUI_PORT} )" | grep -q LISTEN; then
  echo "Hinweis: Open Web UI belegt :${OPENWEBUI_PORT} (erlaubt)." >&2
fi

# 8080 darf nur für Open Web UI genutzt werden – Light Scan
if rg -n --hidden -S ":8080" . --glob '!2.openwebui/**' --glob '!**/.venv/**' --glob '!**/venv*/**' --glob '!project_map/**' >/dev/null 2>&1; then
  echo "ERROR: Nutzung von :8080 außerhalb 2.openwebui gefunden." >&2
  exit 1
fi

# --- Bootstrap ----------------------------------------------------------------
start_venv

# Ensure uvicorn vorhanden, falls benötigt (optional best effort)
if ! "$LOCAL_VENV_LINK/bin/uvicorn" --version >/dev/null 2>&1; then
  "$LOCAL_VENV_LINK/bin/python" -m pip install "uvicorn[standard]" fastapi >/dev/null 2>&1 || true
fi

start_service "opena1" "$DIR_OPENA1" "$OPENA1_PORT"
start_service "kordp"  "$DIR_KORDP"  "$KORDP_PORT"
start_service "archivp" "$DIR_ARCHIVP" "$ARCHIVP_PORT"
start_service "opena2" "$DIR_OPENA2" "$OPENA2_PORT"

echo "→ Warte auf Health-Endpunkte…"
wait_health "$OPENA1_PORT" "opena1"
wait_health "$KORDP_PORT"  "kordp"
wait_health "$ARCHIVP_PORT" "archivp"
wait_health "$OPENA2_PORT" "opena2"

echo "✅ Bootstrap abgeschlossen. Logs: $LOG_DIR/{opena1,kordp,archivp,opena2}.log"
```

```
# 3.opena1_coordinator/main.py
from fastapi import FastAPI, Request
from pathlib import Path
from datetime import datetime
import json, os

app = FastAPI(title="opena1")

BASE_DIR = Path(os.getenv("PORTIER_BASE_DIR",".")).resolve()
ARCHIV_DIR = Path(os.getenv("ARCHIVP_DIR", BASE_DIR / "1.portier_openai" / "archivp"))
ARCHIV_DIR.mkdir(parents=True, exist_ok=True)
INDEX = ARCHIV_DIR / "index.jsonl"

ALLOWED = list(map(int, os.getenv("PORT_WINDOW_START","12344")+" "+os.getenv("PORT_WINDOW_END","12399")).split()))
FORBIDDEN = [8080]

def write_sp(src:str, dst:str, kind:str, body:dict):
    now = datetime.now()
    adir = ARCHIV_DIR / now.strftime("%Y/%m/%d")
    adir.mkdir(parents=True, exist_ok=True)
    spn = int(now.timestamp())
    name = f"SP{spn}_{src}→{dst}_{kind}.json"
    (adir / name).write_text(json.dumps({**body,"strict":True}, ensure_ascii=False, indent=2), encoding="utf-8")
    with (ARCHIV_DIR / "index.jsonl").open("a", encoding="utf-8") as f:
        f.write(json.dumps({"sp":name,"ts":now.isoformat(),"src":src,"dst":dst,"kind":kind,"path":str(adir/name)}, ensure_ascii=False)+"\n")

@app.get("/health")
def health():
    return {"service":"opena1","status":"healthy","base":str(BASE_DIR),"port_policy":{"allowed":ALLOWED,"forbidden":FORBIDDEN},"timestamp":datetime.utcnow().isoformat()+"Z"}

@app.post("/log/opena1")
async def log_opena1(req: Request):
    payload = await req.json()
    write_sp("opena1","archivp","CMD",{"op":"log","payload":payload})
    # … hier ggf. weitere interne Logik …
    write_sp("opena1","archivp","RESP",{"ok":True})
    return {"ok":True}
```

**Begründung:** Einheitliche API-Form, Safepoints pro Aktion (CMD/RESP) in `archivp`.

```
    Erwartet JSON:
    {
      "src": "opena1|kordp|opena2|…",
      "label": "optional/slug",
      "data": <obj> | {"b64":"..."}  # unterstützt Rohdaten via base64
```

### 3) Safepoints

*   Prüfe Ordner: `1.portier_openai/archivp/YYYY/MM/DD/`
    
*   Erwartet pro Request zwei Dateien:
    
    *   `SP<ts>_opena1→archivp_CMD.json` / `…_RESP.json`
        
    *   `SP<ts>_kordp→archivp_CMD.json` / `…_RESP.json`
        
    *   `SP<ts>_archivp→archivp_CMD.json` / `…_RESP.json`
        
    *   `SP<ts>_opena2→archivp_CMD.json` / `…_RESP.json`
        
*   `index.jsonl` enthält eine Zeile pro Safepoint mit Pfadangabe.
    

* * *

## Warum genau so?

*   **Stabile Verträge:** fixe Routen & Namensräume → weniger Brechungen in der Orchestrierung.
    
*   **Prüfbar:** Health liefert auch Policy-Metadaten → Port-Compliance ist sofort sichtbar.
    
*   **Forensik:** Safepoints (CMD/RESP) + Index → vollständige, chronologische Nachverfolgbarkeit.
    
*   **Idempotent:** Die Endpoints sind side-effect-safe genug für wiederholte Orchestrierungsläufe.
