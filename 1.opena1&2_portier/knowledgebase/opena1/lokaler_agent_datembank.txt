🤖 Agent Dashboard - Selbst-regenerierendes Agent-System

Ein fortschrittliches Agent-System mit Long-Time-Gedächtnis, das sich selbst regeneriert und andere Agents orchestriert.
✨ Hauptmerkmale

    🧠 Long-Time-Gedächtnis: Persistente Speicherung aller Gespräche und Agent-Daten
    🔄 Selbst-Regeneration: Automatische Systemwartung und Optimierung
    🎯 Agent-Orchestrierung: Verwaltung und Koordination mehrerer Agents
    🛣️ Pfad-Management: Flexibles Routing-System für Agent-Interaktionen
    💬 Prompt-Verarbeitung: Intelligente Verarbeitung von Benutzer-Eingaben
    📊 Echtzeit-Dashboard: Moderne Web-Oberfläche für Systemüberwachung
    🔌 REST API: Vollständige API für programmatische Interaktionen

🚀 Schnellstart
1. Agent-Server starten

cd dashboard/backend
python agent_server.py

Der Server läuft auf http://localhost:8000
2. Dashboard öffnen

Öffnen Sie http://localhost:8000 in Ihrem Browser
3. Prompts eingeben

    Geben Sie Prompts in das Textfeld ein
    Klicken Sie auf "Prompt senden"
    Nutzen Sie "System regenerieren" für Wartung

"""
Simple API Client for Agent Dashboard
Demonstrates how to interact with the agent system via REST API
"""

import json
import urllib.request
import urllib.parse
import time


class AgentDashboardClient:
    """Client for interacting with the Agent Dashboard API"""

    def __init__(self, base_url="http://localhost:8000"):
        self.base_url = base_url

    def _make_request(self, endpoint, method="GET", data=None):
        """Make HTTP request to the API"""
        url = f"{self.base_url}{endpoint}"

        if method == "GET":
            response = urllib.request.urlopen(url)
        else:
            # POST request
            req_data = json.dumps(data).encode('utf-8') if data else None
            req = urllib.request.Request(
                url,
                data=req_data,
                headers={'Content-Type': 'application/json'}
            )
            response = urllib.request.urlopen(req)

        return json.loads(response.read().decode('utf-8'))

    def get_health(self):
        """Get health status"""
        return self._make_request("/agent/health")

    def list_agents(self):
        """List all agents"""
        return self._make_request("/agent/list")

    def get_stats(self):
        """Get system statistics"""
        return self._make_request("/agent/stats")

    def send_prompt(self, prompt, agent_id=None):
        """Send a prompt to an agent"""
        data = {"prompt": prompt}
        if agent_id:
            data["agent_id"] = agent_id
        return self._make_request("/agent/prompt", "POST", data)

    def create_agent(self, name, agent_type="prompt_processor"):
        """Create a new agent"""
        data = {"name": name, "type": agent_type}
        return self._make_request("/agent/create", "POST", data)

    def regenerate_system(self):
        """Regenerate the agent system"""
        return self._make_request("/agent/regenerate", "POST", {})

    def get_routes(self):
        """Get agent routes"""
        return self._make_request("/agent/routes")

"""
Agent API Routes
FastAPI routes for agent management
"""

import json
from datetime import datetime
from typing import Dict, List, Any, Optional

# Since we can't install FastAPI, we'll create a simple HTTP handler
from http.server import HTTPServer, BaseHTTPRequestHandler
import urllib.parse as urlparse


class AgentAPIHandler(BaseHTTPRequestHandler):
    """
    Simple HTTP handler for agent API endpoints
    """

    def __init__(self, agent_manager, *args, **kwargs):
        self.agent_manager = agent_manager
        super().__init__(*args, **kwargs)

    def do_GET(self):
        """Handle GET requests"""
        parsed_path = urlparse.urlparse(self.path)
        path = parsed_path.path

        if path == '/agent/health':
            self._send_json_response({"status": "healthy", "timestamp": datetime.now().isoformat()})

        elif path == '/agent/list':
            agents = self.agent_manager.list_agents()
            self._send_json_response({"agents": agents})

        elif path == '/agent/stats':
            stats = self.agent_manager.get_system_stats()
            self._send_json_response(stats)

        elif path == '/agent/routes':
            routes = self.agent_manager.get_agent_routes()
            self._send_json_response({"routes": routes})

        elif path.startswith('/agent/') and path.endswith('/conversations'):
            # Extract agent ID from path
            agent_id = path.split('/')[2]
            conversations = self.agent_manager.get_conversation_history(agent_id)
            self._send_json_response({"conversations": conversations})

        else:
            self._send_error_response(404, "Not Found")

    def do_POST(self):
        """Handle POST requests"""
        parsed_path = urlparse.urlparse(self.path)
        path = parsed_path.path

        # Read request body
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length)

        try:
            data = json.loads(post_data.decode('utf-8'))
        except json.JSONDecodeError:
            self._send_error_response(400, "Invalid JSON")
            return

        if path == '/agent/create':
            name = data.get('name', 'NewAgent')
            agent_type = data.get('type', 'prompt_processor')

            try:
                agent_id = self.agent_manager.create_agent(name, agent_type)
                self._send_json_response({
                    "success": True,
                    "agent_id": agent_id,
                    "message": f"Agent '{name}' created successfully"
                })
            except Exception as e:
                self._send_error_response(500, f"Failed to create agent: {str(e)}")

        elif path == '/agent/prompt':
            prompt = data.get('prompt', '')
            agent_id = data.get('agent_id')

            if not prompt:
                self._send_error_response(400, "Prompt is required")
                return

            try:
                result = self.agent_manager.process_prompt(prompt, agent_id)
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to process prompt: {str(e)}")

        elif path == '/agent/task':
            agent_id = data.get('agent_id')
            task = data.get('task', {})

            if not agent_id or not task:
                self._send_error_response(400, "Agent ID and task are required")
                return

            try:
                result = self.agent_manager.execute_agent_task(agent_id, task)
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to execute task: {str(e)}")

        elif path == '/agent/regenerate':
            try:
                result = self.agent_manager.self_regenerate()
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to regenerate: {str(e)}")

        else:
            self._send_error_response(404, "Not Found")

    def _send_json_response(self, data: Dict[str, Any], status_code: int = 200):
        """Send JSON response"""
        self.send_response(status_code)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()

        response_data = json.dumps(data, indent=2, default=str)
        self.wfile.write(response_data.encode('utf-8'))

    def _send_error_response(self, status_code: int, message: str):
        """Send error response"""
        self._send_json_response({
            "error": message,
            "timestamp": datetime.now().isoformat()
        }, status_code)

    def do_OPTIONS(self):
        """Handle OPTIONS requests for CORS"""
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()


def create_agent_api_handler(agent_manager):
    """Create HTTP handler with agent manager"""
    def handler(*args, **kwargs):
        return AgentAPIHandler(agent_manager, *args, **kwargs)
    return handler

"""
Agent Base Class
Base class for all agents in the system
"""

import uuid
import json
from datetime import datetime
from typing import Dict, List, Any, Optional
from abc import ABC, abstractmethod


class AgentBase(ABC):
    """
    Base class for all agents in the system

    Provides common functionality for:
    - Agent identification
    - State management
    - Communication with other agents
    - Memory integration
    """

    def __init__(self, name: str, agent_type: str = "generic"):
        self.id = str(uuid.uuid4())
        self.name = name
        self.agent_type = agent_type
        self.created_at = datetime.now()
        self.last_activity = datetime.now()
        self.state = "inactive"
        self.memory = {}
        self.routes = []
        self.parent_agent = None
        self.child_agents = []

    def get_info(self) -> Dict[str, Any]:
        """Get agent information"""
        return {
            "id": self.id,
            "name": self.name,
            "type": self.agent_type,
            "created_at": self.created_at.isoformat(),
            "last_activity": self.last_activity.isoformat(),
            "state": self.state,
            "memory_size": len(self.memory),
            "routes": self.routes,
            "child_agents": len(self.child_agents)
        }

    def update_activity(self):
        """Update last activity timestamp"""
        self.last_activity = datetime.now()

    def add_memory(self, key: str, value: Any):
        """Add information to agent memory"""
        self.memory[key] = {
            "value": value,
            "timestamp": datetime.now().isoformat()
        }
        self.update_activity()

    def get_memory(self, key: str) -> Any:
        """Retrieve information from agent memory"""
        self.update_activity()
        return self.memory.get(key, {}).get("value")

    def add_route(self, route: str):
        """Add a route/path to this agent"""
        if route not in self.routes:
            self.routes.append(route)
            self.update_activity()

    def set_parent(self, parent_agent: 'AgentBase'):
        """Set parent agent"""
        self.parent_agent = parent_agent
        if self not in parent_agent.child_agents:
            parent_agent.child_agents.append(self)
        self.update_activity()

    def add_child(self, child_agent: 'AgentBase'):
        """Add child agent"""
        if child_agent not in self.child_agents:
            self.child_agents.append(child_agent)
            child_agent.parent_agent = self
        self.update_activity()

    @abstractmethod
    def execute_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Execute a task - must be implemented by subclasses"""
        pass

    @abstractmethod
    def process_prompt(self, prompt: str) -> str:
        """Process a prompt - must be implemented by subclasses"""
        pass

    def serialize(self) -> Dict[str, Any]:
        """Serialize agent state for persistence"""
        return {
            "id": self.id,
            "name": self.name,
            "agent_type": self.agent_type,
            "created_at": self.created_at.isoformat(),
            "last_activity": self.last_activity.isoformat(),
            "state": self.state,
            "memory": self.memory,
            "routes": self.routes,
            "child_agent_ids": [child.id for child in self.child_agents]
        }

    @classmethod
    def deserialize(cls, data: Dict[str, Any]) -> 'AgentBase':
        """Deserialize agent state from persistence"""
        agent = cls(data["name"], data["agent_type"])
        agent.id = data["id"]
        agent.created_at = datetime.fromisoformat(data["created_at"])
        agent.last_activity = datetime.fromisoformat(data["last_activity"])
        agent.state = data["state"]
        agent.memory = data["memory"]
        agent.routes = data["routes"]
        return agent

"""
Agent Manager
Central management system for all agents
"""

import json
from datetime import datetime
from typing import Dict, List, Any, Optional
from pathlib import Path

from .agent_base import AgentBase
from .memory_system import MemorySystem


class PromptAgent(AgentBase):
    """
    Simple prompt processing agent
    """

    def __init__(self, name: str = "PromptAgent"):
        super().__init__(name, "prompt_processor")
        self.state = "active"

    def execute_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Execute a task"""
        task_type = task.get("type", "unknown")

        if task_type == "process_prompt":
            return self._process_prompt_task(task)
        elif task_type == "generate_response":
            return self._generate_response_task(task)
        else:
            return {"error": f"Unknown task type: {task_type}"}

    def _process_prompt_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Process a prompt task"""
        prompt = task.get("prompt", "")
        context = task.get("context", {})

        # Store prompt in memory
        self.add_memory(f"prompt_{datetime.now().isoformat()}", {
            "prompt": prompt,
            "context": context
        })

        # Simple response generation
        response = self.process_prompt(prompt)

        return {
            "success": True,
            "response": response,
            "agent_id": self.id,
            "timestamp": datetime.now().isoformat()
        }

    def _generate_response_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Generate a response task"""
        input_data = task.get("input", "")

        # Process the input
        response = f"Processed input: {input_data}"

        return {
            "success": True,
            "response": response,
            "agent_id": self.id,
            "timestamp": datetime.now().isoformat()
        }

    def process_prompt(self, prompt: str) -> str:
        """Process a prompt and generate response"""
        # Simple prompt processing logic
        prompt_lower = prompt.lower()

        if "hallo" in prompt_lower or "hello" in prompt_lower:
            return f"Hallo! Ich bin {self.name}, ein Agent im Hyperdashboard-System. Wie kann ich helfen?"

        elif "agent" in prompt_lower and "erstellen" in prompt_lower:
            return "Ich kann neue Agents erstellen. Welchen Typ von Agent benötigen Sie?"

        elif "status" in prompt_lower:
            return f"Agent {self.name} ist aktiv. Speicher: {len(self.memory)} Einträge, Routen: {len(self.routes)}"

        elif "hilfe" in prompt_lower or "help" in prompt_lower:
            return """Verfügbare Befehle:
- Agent erstellen
- Status abfragen
- Speicher verwalten
- Routen konfigurieren
- Andere Agents einleiten"""

        else:
            return f"Prompt empfangen: '{prompt}'. Verarbeitet von {self.name} um {datetime.now().strftime('%H:%M:%S')}"


class AgentManager:
    """
    Central management system for all agents

    Features:
    - Agent registration and lifecycle management
    - Agent orchestration and communication
    - Self-regeneration capabilities
    - Path/route management
    - Integration with memory system
    """

    def __init__(self, storage_path: str = "/tmp/agent_system"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)

        self.agents: Dict[str, AgentBase] = {}
        self.memory_system = MemorySystem(str(self.storage_path / "memory"))
        self.agent_registry_file = self.storage_path / "agent_registry.json"

        # Initialize with a default prompt agent
        self._initialize_system()

    def _initialize_system(self):
        """Initialize the agent system with default agents"""
        # Create main prompt processing agent
        main_agent = PromptAgent("MainPromptAgent")
        main_agent.add_route("/prompt")
        main_agent.add_route("/dashboard")

        self.register_agent(main_agent)

        # Load any previously saved agents
        self._load_agent_registry()

    def register_agent(self, agent: AgentBase) -> str:
        """Register a new agent"""
        agent_id = agent.id
        self.agents[agent_id] = agent

        # Store in memory system
        self.memory_system.store_agent_memory(agent_id, {
            "registered_at": datetime.now().isoformat(),
            "agent_info": agent.get_info()
        })

        # Save registry
        self._save_agent_registry()

        return agent_id

    def get_agent(self, agent_id: str) -> Optional[AgentBase]:
        """Get agent by ID"""
        return self.agents.get(agent_id)

    def get_agent_by_name(self, name: str) -> Optional[AgentBase]:
        """Get agent by name"""
        for agent in self.agents.values():
            if agent.name == name:
                return agent
        return None

    def list_agents(self) -> List[Dict[str, Any]]:
        """List all registered agents"""
        return [agent.get_info() for agent in self.agents.values()]

    def remove_agent(self, agent_id: str) -> bool:
        """Remove an agent from the system"""
        if agent_id in self.agents:
            agent = self.agents[agent_id]

            # Remove from parent/child relationships
            if agent.parent_agent:
                agent.parent_agent.child_agents.remove(agent)

            for child in agent.child_agents:
                child.parent_agent = None

            # Remove from registry
            del self.agents[agent_id]

            # Save registry
            self._save_agent_registry()

            return True
        return False

    def create_agent(self, name: str, agent_type: str = "prompt_processor") -> str:
        """Create a new agent"""
        if agent_type == "prompt_processor":
            agent = PromptAgent(name)
        else:
            # For now, default to prompt agent
            agent = PromptAgent(name)

        return self.register_agent(agent)

    def process_prompt(self, prompt: str, agent_id: Optional[str] = None) -> Dict[str, Any]:
        """Process a prompt through the agent system"""
        # Use specific agent or default to main agent
        if agent_id:
            agent = self.get_agent(agent_id)
        else:
            agent = self.get_agent_by_name("MainPromptAgent")

        if not agent:
            return {
                "error": "No suitable agent found",
                "timestamp": datetime.now().isoformat()
            }

        # Process the prompt
        response = agent.process_prompt(prompt)

        # Store conversation in memory
        self.memory_system.store_conversation(agent.id, prompt, response)

"""
Memory System
Long-term memory management for agents
"""

import json
import os
from datetime import datetime
from typing import Dict, List, Any, Optional
from pathlib import Path


class MemorySystem:
    """
    Long-term memory system for agent data persistence

    Features:
    - Persistent storage of agent memories
    - Conversation history
    - Agent relationships
    - Performance metrics
    - Automatic cleanup of old memories
    """

    def __init__(self, storage_path: str = "/tmp/agent_memory"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.memory_file = self.storage_path / "agent_memories.json"
        self.conversation_file = self.storage_path / "conversations.json"
        self.relationships_file = self.storage_path / "relationships.json"
        self.metrics_file = self.storage_path / "metrics.json"

        # Initialize storage files
        self._initialize_storage()

    def _initialize_storage(self):
        """Initialize storage files if they don't exist"""
        for file_path in [self.memory_file, self.conversation_file,
                         self.relationships_file, self.metrics_file]:
            if not file_path.exists():
                self._save_json(file_path, {})

    def _load_json(self, file_path: Path) -> Dict[str, Any]:
        """Load JSON data from file"""
        try:
            with open(file_path, 'r') as f:
                return json.load(f)
        except (FileNotFoundError, json.JSONDecodeError):
            return {}

    def _save_json(self, file_path: Path, data: Dict[str, Any]):
        """Save JSON data to file"""
        with open(file_path, 'w') as f:
            json.dump(data, f, indent=2, default=str)

    def store_agent_memory(self, agent_id: str, memory_data: Dict[str, Any]):
        """Store agent memory data"""
        memories = self._load_json(self.memory_file)

        if agent_id not in memories:
            memories[agent_id] = {
                "created_at": datetime.now().isoformat(),
                "memories": {}
            }

        memories[agent_id]["memories"].update(memory_data)
        memories[agent_id]["last_updated"] = datetime.now().isoformat()

        self._save_json(self.memory_file, memories)

    def get_agent_memory(self, agent_id: str) -> Dict[str, Any]:
        """Retrieve agent memory data"""
        memories = self._load_json(self.memory_file)
        return memories.get(agent_id, {}).get("memories", {})

    def store_conversation(self, agent_id: str, prompt: str, response: str):
        """Store conversation history"""
        conversations = self._load_json(self.conversation_file)

        if agent_id not in conversations:
            conversations[agent_id] = []

        conversation_entry = {
            "timestamp": datetime.now().isoformat(),
            "prompt": prompt,
            "response": response
        }

        conversations[agent_id].append(conversation_entry)

        # Keep only last 100 conversations per agent
        if len(conversations[agent_id]) > 100:
            conversations[agent_id] = conversations[agent_id][-100:]

        self._save_json(self.conversation_file, conversations)

    def get_conversation_history(self, agent_id: str, limit: int = 10) -> List[Dict[str, Any]]:
        """Get conversation history for an agent"""
        conversations = self._load_json(self.conversation_file)
        agent_conversations = conversations.get(agent_id, [])

        # Return last 'limit' conversations
        return agent_conversations[-limit:] if agent_conversations else []

    def store_relationship(self, parent_agent_id: str, child_agent_id: str, relationship_type: str = "parent-child"):
        """Store agent relationship"""
        relationships = self._load_json(self.relationships_file)

        if parent_agent_id not in relationships:
            relationships[parent_agent_id] = {
                "children": [],
                "relationships": {}
            }

        if child_agent_id not in relationships[parent_agent_id]["children"]:
            relationships[parent_agent_id]["children"].append(child_agent_id)

        relationships[parent_agent_id]["relationships"][child_agent_id] = {
            "type": relationship_type,
            "created_at": datetime.now().isoformat()
        }

        self._save_json(self.relationships_file, relationships)

    def get_agent_relationships(self, agent_id: str) -> Dict[str, Any]:
        """Get agent relationships"""
        relationships = self._load_json(self.relationships_file)
        return relationships.get(agent_id, {"children": [], "relationships": {}})

    def store_metric(self, agent_id: str, metric_name: str, value: Any):
        """Store performance metric"""
        metrics = self._load_json(self.metrics_file)

        if agent_id not in metrics:
            metrics[agent_id] = {}

        if metric_name not in metrics[agent_id]:
            metrics[agent_id][metric_name] = []

        metric_entry = {
            "timestamp": datetime.now().isoformat(),
            "value": value
        }

        metrics[agent_id][metric_name].append(metric_entry)

        # Keep only last 1000 metrics per agent per metric type
        if len(metrics[agent_id][metric_name]) > 1000:
            metrics[agent_id][metric_name] = metrics[agent_id][metric_name][-1000:]

        self._save_json(self.metrics_file, metrics)

    def get_metrics(self, agent_id: str) -> Dict[str, List[Dict[str, Any]]]:
        """Get performance metrics for an agent"""
        metrics = self._load_json(self.metrics_file)
        return metrics.get(agent_id, {})

    def get_all_agent_ids(self) -> List[str]:
        """Get all agent IDs that have stored data"""
        all_ids = set()

        # Check all storage files
        for file_path in [self.memory_file, self.conversation_file,
                         self.relationships_file, self.metrics_file]:
            data = self._load_json(file_path)
            all_ids.update(data.keys())

        return list(all_ids)

    def cleanup_old_data(self, days_old: int = 30):
        """Clean up data older than specified days"""
        from datetime import timedelta

        cutoff_date = datetime.now() - timedelta(days=days_old)

        # Clean up conversations
        conversations = self._load_json(self.conversation_file)
        for agent_id, conv_list in conversations.items():
            conversations[agent_id] = [
                conv for conv in conv_list
                if datetime.fromisoformat(conv["timestamp"]) > cutoff_date
            ]
        self._save_json(self.conversation_file, conversations)

        # Clean up metrics
        metrics = self._load_json(self.metrics_file)
        for agent_id, agent_metrics in metrics.items():
            for metric_name, metric_list in agent_metrics.items():
                metrics[agent_id][metric_name] = [
                    metric for metric in metric_list
                    if datetime.fromisoformat(metric["timestamp"]) > cutoff_date
                ]
        self._save_json(self.metrics_file, metrics)

    def export_agent_data(self, agent_id: str) -> Dict[str, Any]:
        """Export all data for a specific agent"""
        return {
            "agent_id": agent_id,
            "memory": self.get_agent_memory(agent_id),
            "conversations": self.get_conversation_history(agent_id, limit=1000),
            "relationships": self.get_agent_relationships(agent_id),
            "metrics": self.get_metrics(agent_id),
            "exported_at": datetime.now().isoformat()
        }

    def get_memory_stats(self) -> Dict[str, Any]:
        """Get memory system statistics"""
        stats = {
            "total_agents": len(self.get_all_agent_ids()),
            "storage_path": str(self.storage_path),
            "files": {}
        }

        # Get file sizes
        for file_path in [self.memory_file, self.conversation_file,
                         self.relationships_file, self.metrics_file]:
            if file_path.exists():
                stats["files"][file_path.name] = {
                    "size_bytes": file_path.stat().st_size,
                    "entries": len(self._load_json(file_path))
                }

        return stats

        self.metrics_file = self.storage_path / "metrics.json"
        self._initialize_storage()

    def _atomic_write(self, path: Path, data: Dict[str, Any]) -> None:
        tmp = path.with_suffix(".tmp")
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, default=str)
            f.flush()
            os.fsync(f.fileno())
        tmp.replace(path)

    def _load_json(self, path: Path) -> Dict[str, Any]:
        if not path.exists():
            return {}
        try:
            with path.open("r", encoding="utf-8") as f:
                return json.load(f)
        except (json.JSONDecodeError, OSError):
            # defekt: safe fallback
            return {}

    def _initialize_storage(self) -> None:
        for p in [self.memory_file, self.conversation_file, self.relationships_file, self.metrics_file]:
            if not p.exists():
                self._atomic_write(p, {})

    # ---------- Memories ---------- #
    def store_agent_memory(self, agent_id: str, memory_data: Dict[str, Any]) -> None:
        memories = self._load_json(self.memory_file)
        if agent_id not in memories:
            memories[agent_id] = {"created_at": now_iso_z(), "memories": {}}
        # Merke: überschreiben bewusst (deterministisch)
        memories[agent_id]["memories"].update(memory_data)
        memories[agent_id]["last_updated"] = now_iso_z()
        self._atomic_write(self.memory_file, memories)

    def get_agent_memory(self, agent_id: str) -> Dict[str, Any]:
        memories = self._load_json(self.memory_file)
        return memories.get(agent_id, {}).get("memories", {})

    # ---------- Conversations ---------- #
    def store_conversation(self, agent_id: str, prompt: str, response: str) -> None:
        conversations = self._load_json(self.conversation_file)
        if agent_id not in conversations:
            conversations[agent_id] = []
        conversations[agent_id].append({"timestamp": now_iso_z(), "prompt": prompt, "response": response})
        # Grenzen: nur letzte 100 Einträge pro Agent halten
        if len(conversations[agent_id]) > 100:
            conversations[agent_id] = conversations[agent_id][-100:]
        self._atomic_write(self.conversation_file, conversations)

    def get_conversation_history(self, agent_id: str, limit: int = 10) -> List[Dict[str, Any]]:
        conversations = self._load_json(self.conversation_file)
        agent_convs = conversations.get(agent_id, [])
        return agent_convs[-limit:] if agent_convs else []

    # ---------- Relationships & Metrics ---------- #
    def store_relationship(self, parent_id: str, child_id: str, relationship_type: str = "parent-child") -> None:
        rels = self._load_json(self.relationships_file)
        if parent_id not in rels:
            rels[parent_id] = {"children": [], "relationships": {}}
        if child_id not in rels[parent_id]["children"]:
            rels[parent_id]["children"].append(child_id)
        rels[parent_id]["relationships"][child_id] = {"type": relationship_type, "created_at": now_iso_z()}
        self._atomic_write(self.relationships_file, rels)

    def get_agent_relationships(self, agent_id: str) -> Dict[str, Any]:
        rels = self._load_json(self.relationships_file)
        return rels.get(agent_id, {"children": [], "relationships": {}})

    def store_metric(self, agent_id: str, metric_name: str, value: Any) -> None:
        metrics = self._load_json(self.metrics_file)
        if agent_id not in metrics:
            metrics[agent_id] = {}
        if metric_name not in metrics[agent_id]:
            metrics[agent_id][metric_name] = []
        metrics[agent_id][metric_name].append({"timestamp": now_iso_z(), "value": value})
        if len(metrics[agent_id][metric_name]) > 1000:
            metrics[agent_id][metric_name] = metrics[agent_id][metric_name][-1000:]
        self._atomic_write(self.metrics_file, metrics)

    def get_metrics(self, agent_id: str) -> Dict[str, Any]:
        metrics = self._load_json(self.metrics_file)
        return metrics.get(agent_id, {})

    # ---------- Utility / Maintenance ---------- #
    def get_all_agent_ids(self) -> List[str]:
        ids = set()
        for file in [self.memory_file, self.conversation_file, self.relationships_file, self.metrics_file]:
            data = self._load_json(file)
            ids.update(data.keys())
        return list(ids)

    def cleanup_old_data(self, days_old: int = 30) -> None:
        cutoff = datetime.now(timezone.utc) - timedelta(days=days_old)
        # Conversations
        conversations = self._load_json(self.conversation_file)
        for aid, convs in list(conversations.items()):
            conversations[aid] = [c for c in convs if datetime.fromisoformat(c["timestamp"]) > cutoff]
        self._atomic_write(self.conversation_file, conversations)
        # Metrics
        metrics = self._load_json(self.metrics_file)
        for aid, agent_metrics in metrics.items():
            for mname, mlist in agent_metrics.items():
                metrics[aid][mname] = [m for m in mlist if datetime.fromisoformat(m["timestamp"]) > cutoff]
        self._atomic_write(self.metrics_file, metrics)

    def export_agent_data(self, agent_id: str) -> Dict[str, Any]:
        return {
            "agent_id": agent_id,
            "memory": self.get_agent_memory(agent_id),
            "conversations": self.get_conversation_history(agent_id, limit=1000),
            "relationships": self.get_agent_relationships(agent_id),
            "metrics": self.get_metrics(agent_id),
            "exported_at": now_iso_z(),
        }

    def get_memory_stats(self) -> Dict[str, Any]:
        stats = {"total_agents": len(self.get_all_agent_ids()), "storage_path": str(self.storage_path), "files": {}}
        for p in [self.memory_file, self.conversation_file, self.relationships_file, self.metrics_file]:
            if p.exists():
                stats["files"][p.name] = {"size_bytes": p.stat().st_size, "entries": len(self._load_json(p))}
        return stats

# Minimaler, produktiver PromptAgent (deterministisch, kein Random/Tests)
class PromptAgent(AgentBase):
    def __init__(self, name: str = "MainPromptAgent"):
        super().__init__(name, "prompt_processor")
        self.state = "active"
        # Start-Routen (produktive Default-Routen)
        self.add_route("/prompt")
        self.add_route("/dashboard")

    def execute_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        ttype = task.get("type", "")
        if ttype == "process_prompt":
            prompt = task.get("prompt", "")
            result = self._process_prompt_task(prompt, task.get("context", {}))
            return result
        elif ttype == "generate_response":
            input_data = task.get("input", "")
            return {"success": True, "response": f"Processed input: {input_data}", "agent_id": self.id, "timestamp": datetime.utcnow().isoformat() + "Z"}
        return {"error": f"Unknown task type: {ttype}"}

    def _process_prompt_task(self, prompt: str, context: Dict[str, Any]) -> Dict[str, Any]:
        # Speicher den Prompt in Agent-Memory
        key = f"prompt_{datetime.utcnow().isoformat()}Z"
        self.add_memory(key, {"prompt": prompt, "context": context})
        response = self.process_prompt(prompt)
        return {"success": True, "response": response, "agent_id": self.id, "timestamp": datetime.utcnow().isoformat() + "Z"}

    def process_prompt(self, prompt: str) -> str:
        p = (prompt or "").lower().strip()
        if not p:
            return "Leerer Prompt empfangen."
        if "hallo" in p or "hello" in p:
            return f"Hallo! Ich bin {self.name}. Wie kann ich helfen?"
        if "status" in p:
            return f"Agent {self.name} aktiv. Speicher-Einträge: {len(self._memory)}, Routen: {len(self.routes)}"
        if "agent erstellen" in p or ("agent" in p and "erstellen" in p):
            return "Nenne mir bitte den Namen des neuen Agents (API: /agent/create)."
        if "hilfe" in p:
            return "Verfügbare Befehle: Agent erstellen, Status, Speicher anzeigen, Routen verwalten."
        # Default: Echo (sicher, deterministisch)
        return f"Prompt empfangen: '{prompt}' - verarbeitet von {self.name}."

class AgentManager:
    """
    Produktionsfähiger AgentManager:
    - Registry persistiert (agent_registry.json)
    - Memory-Integration (MemorySystem)
    - Self-regeneration (keine Tests/Mocks)
    """

    DEFAULT_ALLOWED_PORTS = [12344, 12345, 12346, 12347, 12348, 12349]  # Konform zu Projektregeln
    DEFAULT_FORBIDDEN = [8080]

    def __init__(self, storage_path: str = "/tmp/agent_system"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.agents: Dict[str, AgentBase] = {}
        self.memory_system = MemorySystem(str(self.storage_path / "memory"))
        self.agent_registry_file = self.storage_path / "agent_registry.json"
        self.allowed_ports = list(self.DEFAULT_ALLOWED_PORTS)
        self.forbidden_ports = list(self.DEFAULT_FORBIDDEN)
        self._initialize_system()

    def _initialize_system(self) -> None:
        main = PromptAgent("MainPromptAgent")
        self.register_agent(main)
        # lade bestehende Registry (sicherer Fallback)
        self._load_agent_registry()

    def register_agent(self, agent: AgentBase) -> str:
        self.agents[agent.id] = agent
        # Store minimal registry + memory
        self.memory_system.store_agent_memory(agent.id, {"registered_at": datetime.utcnow().isoformat() + "Z", "agent_info": agent.get_info()})
        self._save_agent_registry()
        return agent.id

    def get_agent(self, agent_id: str) -> Optional[AgentBase]:
        return self.agents.get(agent_id)

    def get_agent_by_name(self, name: str) -> Optional[AgentBase]:
        for a in self.agents.values():
            if a.name == name:
                return a
        return None

    def list_agents(self) -> List[Dict[str, Any]]:
        return [a.get_info() for a in self.agents.values()]

    def remove_agent(self, agent_id: str) -> bool:
        if agent_id in self.agents:
            agent = self.agents.pop(agent_id)
            if agent.parent_agent:
                try:
                    agent.parent_agent.child_agents.remove(agent)
                except ValueError:
                    pass
            for c in agent.child_agents:
                c.parent_agent = None
            self._save_agent_registry()
            return True
        return False

    def create_agent(self, name: str, agent_type: str = "prompt_processor") -> str:
        if agent_type == "prompt_processor":
            agent = PromptAgent(name)
        else:
            agent = PromptAgent(name)
        return self.register_agent(agent)

    def process_prompt(self, prompt: str, agent_id: Optional[str] = None) -> Dict[str, Any]:
        if agent_id:
            agent = self.get_agent(agent_id)
        else:
            agent = self.get_agent_by_name("MainPromptAgent")
        if not agent:
            return {"error": "Kein geeigneter Agent gefunden", "timestamp": datetime.utcnow().isoformat() + "Z"}
        response = agent.process_prompt(prompt)
        self.memory_system.store_conversation(agent.id, prompt, response)
        self.memory_system.store_metric(agent.id, "prompts_processed", 1)
        return {"success": True, "response": response, "agent_id": agent.id, "agent_name": agent.name, "timestamp": datetime.utcnow().isoformat() + "Z"}

    def execute_agent_task(self, agent_id: str, task: Dict[str, Any]) -> Dict[str, Any]:
        agent = self.get_agent(agent_id)
        if not agent:
            return {"error": f"Agent {agent_id} nicht gefunden", "timestamp": datetime.utcnow().isoformat() + "Z"}
        try:
            result = agent.execute_task(task)
            self.memory_system.store_metric(agent_id, "tasks_executed", 1)
            return result
        except Exception as e:
            return {"error": f"Task execution failed: {str(e)}", "agent_id": agent_id, "timestamp": datetime.utcnow().isoformat() + "Z"}

    def get_agent_routes(self) -> Dict[str, Any]:
        return {aid: {"agent_name": a.name, "routes": a.routes} for aid, a in self.agents.items()}

    def add_agent_route(self, agent_id: str, route: str) -> bool:
        a = self.get_agent(agent_id)
        if not a:
            return False
        a.add_route(route)
        self._save_agent_registry()
        return True

    def create_agent_relationship(self, parent_id: str, child_id: str) -> bool:
        parent = self.get_agent(parent_id)
        child = self.get_agent(child_id)
        if parent and child:
            parent.add_child(child)
            self.memory_system.store_relationship(parent_id, child_id)
            return True
        return False

    def get_conversation_history(self, agent_id: str, limit: int = 10) -> List[Dict[str, Any]]:
        return self.memory_system.get_conversation_history(agent_id, limit)

    def get_system_stats(self) -> Dict[str, Any]:
        stats = {
            "total_agents": len(self.agents),
            "active_agents": len([a for a in self.agents.values() if a.state == "active"]),
            "memory_stats": self.memory_system.get_memory_stats(),
            "agent_types": {},
            "total_routes": sum(len(a.routes) for a in self.agents.values()),
        }
        for a in self.agents.values():
            stats["agent_types"][a.agent_type] = stats["agent_types"].get(a.agent_type, 0) + 1
        return stats

    def self_regenerate(self) -> Dict[str, Any]:
        log = []
        for aid, a in list(self.agents.items()):
            if a.state == "inactive":
                a.state = "active"
                log.append(f"Reactivated agent {a.name}")
        prompt_agents = [a for a in self.agents.values() if a.agent_type == "prompt_processor"]
        if not prompt_agents:
            nid = self.create_agent("RegeneratedPromptAgent")
            log.append(f"Created new prompt agent: {nid}")
        self.memory_system.cleanup_old_data(days_old=7)
        log.append("Cleaned up old memory data")
        self._save_agent_registry()
        log.append("Saved agent registry")
        return {"success": True, "regeneration_log": log, "timestamp": datetime.utcnow().isoformat() + "Z"}

    def _save_agent_registry(self) -> None:
        registry = {"agents": {aid: a.serialize() for aid, a in self.agents.items()}, "last_saved": datetime.utcnow().isoformat() + "Z"}
        with self.agent_registry_file.open("w", encoding="utf-8") as f:

        except Exception:

# Port-Policy (statisch gemäß Projektvorgaben)
ALLOWED_PORTS: List[int] = [12344, 12345, 12346, 12347, 12348, 12349]
FORBIDDEN_PORTS: List[int] = [8080]

app = FastAPI(title="Portier Agent Server")
app.include_router(agent_router)

@app.get("/health")
def health():
    return {"service": "portier_agent_server", "status": "healthy", "ports_allowed": ALLOWED_PORTS, "port_forbidden": FORBIDDEN_PORTS}

def _first_free_allowed() -> int:
    for p in ALLOWED_PORTS:
        if p in FORBIDDEN_PORTS:
            continue
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            try:
                s.bind(("127.0.0.1", p))
                return p
            except OSError:
                continue
    # Fallback: letzter erlaubter Port
    return ALLOWED_PORTS[0]

def run():
    port_env = os.getenv("PORTIER_AGENT_PORT")
    if port_env:
        try:
            p = int(port_env)
            if p in FORBIDDEN_PORTS:
                raise RuntimeError(f"Port {p} ist verboten (Policy).")
            if p not in ALLOWED_PORTS:
                raise RuntimeError(f"Port {p} ist nicht in der Allow-Liste.")
            uvicorn.run("agents.agent_server:app", host="127.0.0.1", port=p, log_level="info")
            return
        except Exception as e:
            raise

    chosen = _first_free_allowed()
    uvicorn.run("agents.agent_server:app", host="127.0.0.1", port=chosen, log_level="info")


if __name__ == "__main__":
    run()

10 wichtigste Anforderungen für opena1 (Koordinator)
1) Port- & Netzwerk-Policy durchsetzen

Warum: Security / Konsistenz (Port-Policy ist Projektgesetz).
Datenstruktur: none (Policy-Konstanten).

{
  "allowed_ports": [12344,12345,12346,12347,12348,12349],
  "forbidden_ports": [8080]
}

Ausführung:

    Prüfe beim Start und bei ENV-Override (PORTIER_AGENT_PORT) ob Port in allowed_ports und nicht in forbidden_ports.

    Wenn Bind schlägt fehl => wähle nächstes freies aus Liste; logge Fehler ins Safepoint.

    Implementiere in agent_server.run() und als Guard in allen network-bind-Funktionen.

2) Eingabe-/Ausgabe-Schema & Validierung (7.1/7.2-konform)

Warum: Interoperabilität, Auditierbarkeit, Fehlervermeidung.
Datenstruktur (Request/Response):

// Request
{"request_id":"<hex-uuid>","timestamp":"<ISO>","command":"PROCESS_PROMPT","payload":{"prompt":"..."}}
// Response
{"request_id":"...","timestamp":"<ISO>","source":"opena1","operation":{...},"status":"ok|error","strict":true}

Ausführung:

    Validate mittels Pydantic/JSON-Schema an API-Boundary.

    Alle internen Antworten müssen source":"opena1" und strict:true enthalten.

    Fehlerinhalte in error Feld konsistent zurückgeben.

3) Safepoint-/Audit-Logging (Archivator flow)

Warum: Revisionssichere Nachvollziehbarkeit (SP-JSON Dateien).
Datenstruktur (Safepoint):

{
  "sp":"SP{ts}_opena1→opena2_CMD.json",
  "ts":"2025-10-14T12:00:00Z",
  "src":"opena1","dst":"opena2",
  "kind":"CMD|RESP|ERR",
  "body":{ ... }
}

Ausführung:

    Vor jeder weiterreichenden Aktion: write_safepoint("opena1","opena2","CMD", body).

    Nach Antwort von opena2 / Tool: Safepoint RESP schreiben.

    Speichere unter /archivp/YYYY/MM/DD/ und indexiere in index.jsonl.

4) Agent-Registry & Routing (Agenten-Katalog)

Warum: Koordinator muss wissen, welche Agents existieren, Routes, Zustände.
Datenstruktur (Registry-Eintrag):

{
  "id":"uuid",
  "name":"MainPromptAgent",
  "type":"prompt_processor",
  "state":"active|inactive",
  "routes":["/prompt","/dashboard"],
  "created_at":"ISO"
}

Ausführung:

    Implementiere persistente Registry (agent_registry.json) mit atomic write.

    API: GET /agent/list, GET /agent/routes.

    Routing-Engine: einfache Rule-Table (intent → agent_id). Verwende deterministische Matching-Regeln.

5) Conversation RAG / Memory handoff zu opena2 (Archivator)

Warum: Langzeitgedächtnis ausgelagert; opena1 routet / validiert.
Datenstruktur (Handoff):

{"agent_id":"uuid","conversation":[{"prompt":"..","response":"..","ts":".."}],"meta":{"source":"opena1"}}

Ausführung:

    Beim process_prompt: write_safepoint(...CMD) → POST zu opena2 /store/conversation → logge RESP Safepoint.

    Tokenize/normalize Kontexte vor Übergabe.

    Bei Fehlern: retry policy (3x) mit backoff; alle Versuche safepointen.

6) Request Authorization & RBAC (minimal)

Warum: Nur berechtigte Clients dürfen kritische Aktionen (create agent, regenerate).
Datenstruktur (Token/Role):

{"client_id":"xxx","role":"admin|user|viewer","issued_at":"ISO"}

Ausführung:

    API-Middleware prüft JWT/HToken; mappt role zu erlaubten Endpoints.

    Admin-Action (z.B. /agent/create, /agent/regenerate) nur für role: admin.

7) Input Sanitization & Prompt Safety

Datenstruktur: keine, aber Regeln konfigurierbar:

{"deny_patterns":["{{","}}","<script>"],"max_length":20000}

Ausführung:

8) Task Queue & Worker-Separation (sync vs async)

Warum: Koordinator skalierbar; langlaufende Tasks nicht blockieren.
Datenstruktur (Task):

{"task_id":"uuid","type":"process_prompt|task_exec","agent_id":"uuid","payload":{}, "status":"queued|running|done|failed"}

#!/usr/bin/env bash
set -euo pipefail
# tools/setup_python_env.sh
# Erstellt eine lokale Python-Umgebung via pyenv (falls nicht installiert) und eine venv.

PROJECT_ROOT="$(pwd)"
VENV_NAME="venv313"
PYTHON_VERSION="${1:-3.11.4}"   # optionales Argument: gewünschte Python-Version

echo "Project root: $PROJECT_ROOT"
echo "Gewünschte Python-Version: $PYTHON_VERSION"

# 1) Prüfe aktuelles Python
echo "-- aktuelles system python --"
python3 --version || true

# 2) Installiere pyenv falls nicht vorhanden
if ! command -v pyenv >/dev/null 2>&1; then
  echo "pyenv nicht gefunden -> installiere pyenv (dies kann ein paar Minuten dauern)"
  curl https://pyenv.run | bash
  export PATH="$HOME/.pyenv/bin:$PATH"
  eval "$(pyenv init -)"
  eval "$(pyenv virtualenv-init -)"
else
  echo "pyenv vorhanden"
fi

# 3) Stelle sicher, dass pyenv in dieser Session verfügbar ist
export PATH="$HOME/.pyenv/bin:$PATH"
eval "$(pyenv init -)"
eval "$(pyenv virtualenv-init -)"

# 4) Installiere gewünschte Python-Version, falls noch nicht vorhanden
if ! pyenv versions --bare | grep -q "^${PYTHON_VERSION}\$"; then
  echo "Installiere Python $PYTHON_VERSION via pyenv..."
  pyenv install "$PYTHON_VERSION"
else
  echo "Python $PYTHON_VERSION bereits installiert"
fi

# 5) Setze lokale Python-Version für dieses Verzeichnis
pyenv local "$PYTHON_VERSION"

# 6) Erstelle venv (im Projekt-Root)
if [ -d "${VENV_NAME}" ]; then
  echo "Venv ${VENV_NAME} existiert bereits -> benutze vorhandene Umgebung"
else
  echo "Erstelle venv ${VENV_NAME} mit python $(python --version)"
  python -m venv "${VENV_NAME}"
fi

# 7) Aktiviere venv für diese Session
# shellcheck disable=SC1090
source "${VENV_NAME}/bin/activate"
echo "Venv aktiviert: $(which python) $(python --version)"

# 8) Upgrade pip + install requirements
python -m pip install --upgrade pip setuptools wheel
if [ -f "requirements.txt" ]; then
  pip install -r requirements.txt
else
  echo "requirements.txt nicht gefunden — überspringe pip install"
fi

# 9) Optional: Tests ausführen
if command -v pytest >/dev/null 2>&1; then
  echo "Führe pytest aus..."
  pytest -q || echo "Tests: einige Tests fehlgeschlagen (siehe pytest output)"
else
  echo "pytest nicht installiert, überspringe Tests"
fi

echo "Setup abgeschlossen. Aktivieren: source ${VENV_NAME}/bin/activate"
echo "Starte den Agent-Server (Beispiel): cd dashboard/backend && python3 agent_server.py"

#!/usr/bin/env python3
# dashboard/backend/agent_server.py
# Minimal, self-contained Agent Server for local development.
# - FastAPI app exposing the basic endpoints used by the dashboard and CLI.
# - In-memory AgentManager with optional simple JSON persistence to ./data/agent_registry.json
# - Single-file drop-in when dashboard/backend is empty.
#
# Usage:
#   uvicorn agent_server:app --host 127.0.0.1 --port 12344 --reload
# or run directly:
#   python3 agent_server.py
#
# Port recommendation: default 12344 (do not use 8080).

from __future__ import annotations
import json
import uuid
import threading
from datetime import datetime
from pathlib import Path
from typing import Dict, List, Optional, Any

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

# ---- Config ----
DEFAULT_PORT = int(__import__("os").environ.get("PORT", "12344"))
DATA_DIR = Path(__file__).parent / "data"
REGISTRY_FILE = DATA_DIR / "agent_registry.json"
ALLOWED_PORTS = list(range(12344, 12400))
FORBIDDEN_PORTS = [8080]

# ---- Ensure data dir ----
DATA_DIR.mkdir(parents=True, exist_ok=True)


# ---- Models ----
class CreateAgentRequest(BaseModel):
    name: str = Field(..., min_length=1)
    type: Optional[str] = "prompt_processor"


class PromptRequest(BaseModel):
    prompt: str = Field(..., min_length=1)
    agent_id: Optional[str] = None
    context: Optional[Dict[str, Any]] = None


class GenericResponse(BaseModel):
    success: bool
    message: Optional[str] = None
    data: Optional[Any] = None


# ---- Simple Agent & Manager ----
class SimpleAgent:
    def __init__(self, name: str, agent_type: str = "prompt_processor"):
        self.id = str(uuid.uuid4())
        self.name = name
        self.agent_type = agent_type
        self.created_at = datetime.utcnow().isoformat()
        self.last_activity = self.created_at
        self.state = "active"
        self.memory: Dict[str, Any] = {}
        self.routes: List[str] = []

    def get_info(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "type": self.agent_type,
            "created_at": self.created_at,
            "last_activity": self.last_activity,
            "state": self.state,
            "memory_size": len(self.memory),
            "routes": self.routes,
        }

    def process_prompt(self, prompt: str) -> str:
        self.last_activity = datetime.utcnow().isoformat()
        # Minimal, deterministic fake processing (no external LLM calls).
        p = prompt.strip()
        if not p:
            return "Empty prompt."
        lower = p.lower()
        if "hallo" in lower or "hello" in lower:
            return f"Hallo — ich bin {self.name}. Wie kann ich helfen?"
        if "status" in lower:
            return f"Agent {self.name} ist {self.state}. Speicher: {len(self.memory)} Einträge."
        if "hilfe" in lower or "help" in lower:
            return (
                "Verfügbare Befehle: 'Agent erstellen', 'Status', 'Hilfe', oder beliebige Prompt-Eingabe."
            )
        # generic echo with timestamp
        ts = datetime.utcnow().strftime("%Y-%m-%d %H:%M:%SZ")
        # store to memory
        self.memory[f"prompt_{ts}"] = {"prompt": p, "ts": ts}
        return f"Antwort ({ts}): {p}"


class AgentManager:
    def __init__(self, persist_file: Optional[Path] = REGISTRY_FILE):
        self._lock = threading.RLock()
        self.agents: Dict[str, SimpleAgent] = {}
        self.persist_file = persist_file
        self._load_registry()
        if not any(a.agent_type == "prompt_processor" for a in self.agents.values()):
            # create default main agent
            a = SimpleAgent("MainPromptAgent", "prompt_processor")
            a.routes = ["/prompt", "/dashboard"]
            self.register_agent(a)

    def register_agent(self, agent: SimpleAgent) -> str:
        with self._lock:
            self.agents[agent.id] = agent
            self._save_registry()
            return agent.id

    def create_agent(self, name: str, agent_type: str = "prompt_processor") -> str:
        agent = SimpleAgent(name, agent_type)
        return self.register_agent(agent)

    def list_agents(self) -> List[Dict[str, Any]]:
        with self._lock:
            return [a.get_info() for a in self.agents.values()]

    def get_agent(self, agent_id: str) -> Optional[SimpleAgent]:
        return self.agents.get(agent_id)

    def get_agent_by_name(self, name: str) -> Optional[SimpleAgent]:
        for a in self.agents.values():
            if a.name == name:
                return a
        return None

    def process_prompt(self, prompt: str, agent_id: Optional[str] = None) -> Dict[str, Any]:
        with self._lock:
            agent = self.get_agent(agent_id) if agent_id else self.get_agent_by_name("MainPromptAgent")
            if not agent:
                return {"success": False, "message": "No suitable agent found"}
            response = agent.process_prompt(prompt)
            self._save_registry()
            return {"success": True, "response": response, "agent_id": agent.id, "agent_name": agent.name}

    def get_system_stats(self) -> Dict[str, Any]:
        with self._lock:
            stats = {
                "total_agents": len(self.agents),
                "active_agents": len([a for a in self.agents.values() if a.state == "active"]),
                "total_routes": sum(len(a.routes) for a in self.agents.values()),
            }
            return stats

    def self_regenerate(self) -> Dict[str, Any]:
        with self._lock:
            log: List[str] = []
            # Reactivate inactive agents
            for a in self.agents.values():
                if a.state != "active":
                    a.state = "active"
                    log.append(f"Reactivated {a.name}")
            # Ensure prompt agent exists
            if not any(a.agent_type == "prompt_processor" for a in self.agents.values()):
                new_id = self.create_agent("RegeneratedPromptAgent")
                log.append(f"Created prompt agent {new_id}")
            self._save_registry()
            log.append("Saved registry")
            return {"success": True, "regeneration_log": log}

    # Persistence (very small, atomic write)
    def _save_registry(self):
        if not self.persist_file:
            return
        tmp = self.persist_file.with_suffix(".tmp")
        payload = {
            "agents": {aid: self._serialize_agent(a) for aid, a in self.agents.items()},
            "last_saved": datetime.utcnow().isoformat(),
        }
        tmp.parent.mkdir(parents=True, exist_ok=True)
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(payload, f, indent=2, ensure_ascii=False)
        tmp.replace(self.persist_file)

    def _load_registry(self):
        if not self.persist_file or not self.persist_file.exists():
            return
        try:
            with self.persist_file.open("r", encoding="utf-8") as f:
                payload = json.load(f)
            agents = payload.get("agents", {})
            for aid, ad in agents.items():
                a = SimpleAgent(ad.get("name", f"agent-{aid}"), ad.get("type", "generic"))
                a.id = aid
                a.created_at = ad.get("created_at", a.created_at)
                a.last_activity = ad.get("last_activity", a.last_activity)
                a.state = ad.get("state", a.state)
                a.memory = ad.get("memory", {})
                a.routes = ad.get("routes", [])
                self.agents[aid] = a
        except Exception:
            # ignore load errors, start fresh
            return

    @staticmethod
    def _serialize_agent(a: SimpleAgent) -> Dict[str, Any]:
        return {
            "id": a.id,
            "name": a.name,
            "type": a.agent_type,
            "created_at": a.created_at,
            "last_activity": a.last_activity,
            "state": a.state,
            "memory": a.memory,
            "routes": a.routes,
        }


# ---- App ----
app = FastAPI(title="Agent Server (minimal)", version="0.1")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["GET", "POST", "OPTIONS"],
    allow_headers=["*"],
)

manager = AgentManager()


@app.get("/agent/health", response_model=Dict[str, Any])
def health():
    return {
        "status": "healthy",
        "timestamp": datetime.utcnow().isoformat(),
        "allowed_ports": ALLOWED_PORTS,
        "forbidden_ports": FORBIDDEN_PORTS,
    }


@app.get("/agent/list", response_model=List[Dict[str, Any]])
def list_agents():
    return manager.list_agents()


@app.get("/agent/stats", response_model=Dict[str, Any])
def stats():
    return manager.get_system_stats()


@app.get("/agent/routes", response_model=Dict[str, Any])
def routes():
    with manager._lock:
        return {
            aid: {"agent_name": a.name, "routes": a.routes}
            for aid, a in manager.agents.items()
        }


@app.post("/agent/create", response_model=GenericResponse)
def create_agent(req: CreateAgentRequest):
    try:
        aid = manager.create_agent(req.name, req.type)
        return GenericResponse(success=True, message="Agent created", data={"agent_id": aid})
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/agent/prompt", response_model=Dict[str, Any])
def prompt(req: PromptRequest):
    if not req.prompt:
        raise HTTPException(status_code=400, detail="prompt required")
    res = manager.process_prompt(req.prompt, req.agent_id)
    return res


@app.post("/agent/regenerate", response_model=GenericResponse)
def regenerate():
    try:
        res = manager.self_regenerate()
        return GenericResponse(success=True, message="Regenerated", data=res)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


# small convenience endpoints for conversation retrieval
@app.get("/agent/{agent_id}/conversations", response_model=List[Dict[str, Any]])
def get_conversations(agent_id: str, limit: int = 10):
    a = manager.get_agent(agent_id)
    if not a:
        raise HTTPException(status_code=404, detail="agent not found")
    convs = []
    # memory keys are stored as prompt_{ts}; return latest entries
    for k, v in list(a.memory.items())[-limit:]:
        convs.append({"key": k, "entry": v})
    return convs
