🤖 Agent Dashboard - Selbst-regenerierendes Agent-System

Ein fortschrittliches Agent-System mit Long-Time-Gedächtnis, das sich selbst regeneriert und andere Agents orchestriert.
✨ Hauptmerkmale

    🧠 Long-Time-Gedächtnis: Persistente Speicherung aller Gespräche und Agent-Daten
    🔄 Selbst-Regeneration: Automatische Systemwartung und Optimierung
    🎯 Agent-Orchestrierung: Verwaltung und Koordination mehrerer Agents
    🛣️ Pfad-Management: Flexibles Routing-System für Agent-Interaktionen
    💬 Prompt-Verarbeitung: Intelligente Verarbeitung von Benutzer-Eingaben
    📊 Echtzeit-Dashboard: Moderne Web-Oberfläche für Systemüberwachung
    🔌 REST API: Vollständige API für programmatische Interaktionen

🚀 Schnellstart
1. Agent-Server starten

cd dashboard/backend
python agent_server.py

Der Server läuft auf http://localhost:8000
2. Dashboard öffnen

Öffnen Sie http://localhost:8000 in Ihrem Browser
3. Prompts eingeben

    Geben Sie Prompts in das Textfeld ein
    Klicken Sie auf "Prompt senden"
    Nutzen Sie "System regenerieren" für Wartung

"""
Simple API Client for Agent Dashboard
Demonstrates how to interact with the agent system via REST API
"""

import json
import urllib.request
import urllib.parse
import time


class AgentDashboardClient:
    """Client for interacting with the Agent Dashboard API"""

    def __init__(self, base_url="http://localhost:8000"):
        self.base_url = base_url

    def _make_request(self, endpoint, method="GET", data=None):
        """Make HTTP request to the API"""
        url = f"{self.base_url}{endpoint}"

        if method == "GET":
            response = urllib.request.urlopen(url)
        else:
            # POST request
            req_data = json.dumps(data).encode('utf-8') if data else None
            req = urllib.request.Request(
                url,
                data=req_data,
                headers={'Content-Type': 'application/json'}
            )
            response = urllib.request.urlopen(req)

        return json.loads(response.read().decode('utf-8'))

    def get_health(self):
        """Get health status"""
        return self._make_request("/agent/health")

    def list_agents(self):
        """List all agents"""
        return self._make_request("/agent/list")

    def get_stats(self):
        """Get system statistics"""
        return self._make_request("/agent/stats")

    def send_prompt(self, prompt, agent_id=None):
        """Send a prompt to an agent"""
        data = {"prompt": prompt}
        if agent_id:
            data["agent_id"] = agent_id
        return self._make_request("/agent/prompt", "POST", data)

    def create_agent(self, name, agent_type="prompt_processor"):
        """Create a new agent"""
        data = {"name": name, "type": agent_type}
        return self._make_request("/agent/create", "POST", data)

    def regenerate_system(self):
        """Regenerate the agent system"""
        return self._make_request("/agent/regenerate", "POST", {})

    def get_routes(self):
        """Get agent routes"""
        return self._make_request("/agent/routes")

"""
Agent API Routes
FastAPI routes for agent management
"""

import json
from datetime import datetime
from typing import Dict, List, Any, Optional

# Since we can't install FastAPI, we'll create a simple HTTP handler
from http.server import HTTPServer, BaseHTTPRequestHandler
import urllib.parse as urlparse


class AgentAPIHandler(BaseHTTPRequestHandler):
    """
    Simple HTTP handler for agent API endpoints
    """

    def __init__(self, agent_manager, *args, **kwargs):
        self.agent_manager = agent_manager
        super().__init__(*args, **kwargs)

    def do_GET(self):
        """Handle GET requests"""
        parsed_path = urlparse.urlparse(self.path)
        path = parsed_path.path

        if path == '/agent/health':
            self._send_json_response({"status": "healthy", "timestamp": datetime.now().isoformat()})

        elif path == '/agent/list':
            agents = self.agent_manager.list_agents()
            self._send_json_response({"agents": agents})

        elif path == '/agent/stats':
            stats = self.agent_manager.get_system_stats()
            self._send_json_response(stats)

        elif path == '/agent/routes':
            routes = self.agent_manager.get_agent_routes()
            self._send_json_response({"routes": routes})

        elif path.startswith('/agent/') and path.endswith('/conversations'):
            # Extract agent ID from path
            agent_id = path.split('/')[2]
            conversations = self.agent_manager.get_conversation_history(agent_id)
            self._send_json_response({"conversations": conversations})

        else:
            self._send_error_response(404, "Not Found")

    def do_POST(self):
        """Handle POST requests"""
        parsed_path = urlparse.urlparse(self.path)
        path = parsed_path.path

        # Read request body
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length)

        try:
            data = json.loads(post_data.decode('utf-8'))
        except json.JSONDecodeError:
            self._send_error_response(400, "Invalid JSON")
            return

        if path == '/agent/create':
            name = data.get('name', 'NewAgent')
            agent_type = data.get('type', 'prompt_processor')

            try:
                agent_id = self.agent_manager.create_agent(name, agent_type)
                self._send_json_response({
                    "success": True,
                    "agent_id": agent_id,
                    "message": f"Agent '{name}' created successfully"
                })
            except Exception as e:
                self._send_error_response(500, f"Failed to create agent: {str(e)}")

        elif path == '/agent/prompt':
            prompt = data.get('prompt', '')
            agent_id = data.get('agent_id')

            if not prompt:
                self._send_error_response(400, "Prompt is required")
                return

            try:
                result = self.agent_manager.process_prompt(prompt, agent_id)
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to process prompt: {str(e)}")

        elif path == '/agent/task':
            agent_id = data.get('agent_id')
            task = data.get('task', {})

            if not agent_id or not task:
                self._send_error_response(400, "Agent ID and task are required")
                return

            try:
                result = self.agent_manager.execute_agent_task(agent_id, task)
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to execute task: {str(e)}")

        elif path == '/agent/regenerate':
            try:
                result = self.agent_manager.self_regenerate()
                self._send_json_response(result)
            except Exception as e:
                self._send_error_response(500, f"Failed to regenerate: {str(e)}")

        else:
            self._send_error_response(404, "Not Found")

    def _send_json_response(self, data: Dict[str, Any], status_code: int = 200):
        """Send JSON response"""
        self.send_response(status_code)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()

        response_data = json.dumps(data, indent=2, default=str)
        self.wfile.write(response_data.encode('utf-8'))

    def _send_error_response(self, status_code: int, message: str):
        """Send error response"""
        self._send_json_response({
            "error": message,
            "timestamp": datetime.now().isoformat()
        }, status_code)

    def do_OPTIONS(self):
        """Handle OPTIONS requests for CORS"""
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()


def create_agent_api_handler(agent_manager):
    """Create HTTP handler with agent manager"""
    def handler(*args, **kwargs):
        return AgentAPIHandler(agent_manager, *args, **kwargs)
    return handler

"""
Agent Base Class
Base class for all agents in the system
"""

import uuid
import json
from datetime import datetime
from typing import Dict, List, Any, Optional
from abc import ABC, abstractmethod


class AgentBase(ABC):
    """
    Base class for all agents in the system

    Provides common functionality for:
    - Agent identification
    - State management
    - Communication with other agents
    - Memory integration
    """

    def __init__(self, name: str, agent_type: str = "generic"):
        self.id = str(uuid.uuid4())
        self.name = name
        self.agent_type = agent_type
        self.created_at = datetime.now()
        self.last_activity = datetime.now()
        self.state = "inactive"
        self.memory = {}
        self.routes = []
        self.parent_agent = None
        self.child_agents = []

    def get_info(self) -> Dict[str, Any]:
        """Get agent information"""
        return {
            "id": self.id,
            "name": self.name,
            "type": self.agent_type,
            "created_at": self.created_at.isoformat(),
            "last_activity": self.last_activity.isoformat(),
            "state": self.state,
            "memory_size": len(self.memory),
            "routes": self.routes,
            "child_agents": len(self.child_agents)
        }

    def update_activity(self):
        """Update last activity timestamp"""
        self.last_activity = datetime.now()

    def add_memory(self, key: str, value: Any):
        """Add information to agent memory"""
        self.memory[key] = {
            "value": value,
            "timestamp": datetime.now().isoformat()
        }
        self.update_activity()

    def get_memory(self, key: str) -> Any:
        """Retrieve information from agent memory"""
        self.update_activity()
        return self.memory.get(key, {}).get("value")

    def add_route(self, route: str):
        """Add a route/path to this agent"""
        if route not in self.routes:
            self.routes.append(route)
            self.update_activity()

    def set_parent(self, parent_agent: 'AgentBase'):
        """Set parent agent"""
        self.parent_agent = parent_agent
        if self not in parent_agent.child_agents:
            parent_agent.child_agents.append(self)
        self.update_activity()

    def add_child(self, child_agent: 'AgentBase'):
        """Add child agent"""
        if child_agent not in self.child_agents:
            self.child_agents.append(child_agent)
            child_agent.parent_agent = self
        self.update_activity()

    @abstractmethod
    def execute_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Execute a task - must be implemented by subclasses"""
        pass

    @abstractmethod
    def process_prompt(self, prompt: str) -> str:
        """Process a prompt - must be implemented by subclasses"""
        pass

    def serialize(self) -> Dict[str, Any]:
        """Serialize agent state for persistence"""
        return {
            "id": self.id,
            "name": self.name,
            "agent_type": self.agent_type,
            "created_at": self.created_at.isoformat(),
            "last_activity": self.last_activity.isoformat(),
            "state": self.state,
            "memory": self.memory,
            "routes": self.routes,
            "child_agent_ids": [child.id for child in self.child_agents]
        }

    @classmethod
    def deserialize(cls, data: Dict[str, Any]) -> 'AgentBase':
        """Deserialize agent state from persistence"""
        agent = cls(data["name"], data["agent_type"])
        agent.id = data["id"]
        agent.created_at = datetime.fromisoformat(data["created_at"])
        agent.last_activity = datetime.fromisoformat(data["last_activity"])
        agent.state = data["state"]
        agent.memory = data["memory"]
        agent.routes = data["routes"]
        return agent

"""
Agent Manager
Central management system for all agents
"""

import json
from datetime import datetime
from typing import Dict, List, Any, Optional
from pathlib import Path

from .agent_base import AgentBase
from .memory_system import MemorySystem


class PromptAgent(AgentBase):
    """
    Simple prompt processing agent
    """

    def __init__(self, name: str = "PromptAgent"):
        super().__init__(name, "prompt_processor")
        self.state = "active"

    def execute_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Execute a task"""
        task_type = task.get("type", "unknown")

        if task_type == "process_prompt":
            return self._process_prompt_task(task)
        elif task_type == "generate_response":
            return self._generate_response_task(task)
        else:
            return {"error": f"Unknown task type: {task_type}"}

    def _process_prompt_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Process a prompt task"""
        prompt = task.get("prompt", "")
        context = task.get("context", {})

        # Store prompt in memory
        self.add_memory(f"prompt_{datetime.now().isoformat()}", {
            "prompt": prompt,
            "context": context
        })

        # Simple response generation
        response = self.process_prompt(prompt)

        return {
            "success": True,
            "response": response,
            "agent_id": self.id,
            "timestamp": datetime.now().isoformat()
        }

    def _generate_response_task(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Generate a response task"""
        input_data = task.get("input", "")

        # Process the input
        response = f"Processed input: {input_data}"

        return {
            "success": True,
            "response": response,
            "agent_id": self.id,
            "timestamp": datetime.now().isoformat()
        }

    def process_prompt(self, prompt: str) -> str:
        """Process a prompt and generate response"""
        # Simple prompt processing logic
        prompt_lower = prompt.lower()

        if "hallo" in prompt_lower or "hello" in prompt_lower:
            return f"Hallo! Ich bin {self.name}, ein Agent im Hyperdashboard-System. Wie kann ich helfen?"

        elif "agent" in prompt_lower and "erstellen" in prompt_lower:
            return "Ich kann neue Agents erstellen. Welchen Typ von Agent benötigen Sie?"

        elif "status" in prompt_lower:
            return f"Agent {self.name} ist aktiv. Speicher: {len(self.memory)} Einträge, Routen: {len(self.routes)}"

        elif "hilfe" in prompt_lower or "help" in prompt_lower:
            return """Verfügbare Befehle:
- Agent erstellen
- Status abfragen
- Speicher verwalten
- Routen konfigurieren
- Andere Agents einleiten"""

        else:
            return f"Prompt empfangen: '{prompt}'. Verarbeitet von {self.name} um {datetime.now().strftime('%H:%M:%S')}"


class AgentManager:
    """
    Central management system for all agents

    Features:
    - Agent registration and lifecycle management
    - Agent orchestration and communication
    - Self-regeneration capabilities
    - Path/route management
    - Integration with memory system
    """

    def __init__(self, storage_path: str = "/tmp/agent_system"):
        self.storage_path = Path(storage_path)
        self.storage_path.mkdir(parents=True, exist_ok=True)

        self.agents: Dict[str, AgentBase] = {}
        self.memory_system = MemorySystem(str(self.storage_path / "memory"))
        self.agent_registry_file = self.storage_path / "agent_registry.json"

        # Initialize with a default prompt agent
        self._initialize_system()

    def _initialize_system(self):
        """Initialize the agent system with default agents"""
        # Create main prompt processing agent
        main_agent = PromptAgent("MainPromptAgent")
        main_agent.add_route("/prompt")
        main_agent.add_route("/dashboard")

        self.register_agent(main_agent)

        # Load any previously saved agents
        self._load_agent_registry()

    def register_agent(self, agent: AgentBase) -> str:
        """Register a new agent"""
        agent_id = agent.id
        self.agents[agent_id] = agent

        # Store in memory system
        self.memory_system.store_agent_memory(agent_id, {
            "registered_at": datetime.now().isoformat(),
            "agent_info": agent.get_info()
        })

        # Save registry
        self._save_agent_registry()

        return agent_id

    def get_agent(self, agent_id: str) -> Optional[AgentBase]:
        """Get agent by ID"""
        return self.agents.get(agent_id)

    def get_agent_by_name(self, name: str) -> Optional[AgentBase]:
        """Get agent by name"""
        for agent in self.agents.values():
            if agent.name == name:
                return agent
        return None

    def list_agents(self) -> List[Dict[str, Any]]:
        """List all registered agents"""
        return [agent.get_info() for agent in self.agents.values()]

    def remove_agent(self, agent_id: str) -> bool:
        """Remove an agent from the system"""
        if agent_id in self.agents:
            agent = self.agents[agent_id]

            # Remove from parent/child relationships
            if agent.parent_agent:
                agent.parent_agent.child_agents.remove(agent)

            for child in agent.child_agents:
                child.parent_agent = None

            # Remove from registry
            del self.agents[agent_id]

            # Save registry
            self._save_agent_registry()

            return True
        return False

    def create_agent(self, name: str, agent_type: str = "prompt_processor") -> str:
        """Create a new agent"""
        if agent_type == "prompt_processor":
            agent = PromptAgent(name)
        else:
            # For now, default to prompt agent
            agent = PromptAgent(name)

        return self.register_agent(agent)

    def process_prompt(self, prompt: str, agent_id: Optional[str] = None) -> Dict[str, Any]:
        """Process a prompt through the agent system"""
        # Use specific agent or default to main agent
        if agent_id:
            agent = self.get_agent(agent_id)
        else:
            agent = self.get_agent_by_name("MainPromptAgent")

        if not agent:
            return {
                "error": "No suitable agent found",
                "timestamp": datetime.now().isoformat()
            }

        # Process the prompt
        response = agent.process_prompt(prompt)

        # Store conversation in memory
        self.memory_system.store_conversation(agent.id, prompt, response)

Suche nach verfügbaren Tools abgeschlossen

Sauber — Docker läuft jetzt. Logs zeigen, dass dockerd mehrfach sauber gestartet hat, API listen on /run/docker.sock meldet Erfolg. Die früheren Timeouts (SIGKILL) deuten auf manuelle Stops oder auf einen Prozess, der nicht schnell genug runterfährt — nicht unbedingt fatal, nur noisy. Die Warnungen zu „stale sandbox / unable to complete atomic operation“ sind typisch, wenn alte Container/Netzwerke Reste hinterlassen haben.

Ich mache’s kurz, praktisch und sicher — drei kleine Prüf- und Aufräumschritte, die du jetzt ausführen kannst, plus Hinweise, was sie bewirken.
1) Schnell-Checks (gibt kurz Status)

Kopiere ein-/zweizeilig in dein Terminal:

docker --version
docker info --format '{{.ServerVersion}} {{.OperatingSystem}} {{.NCPU}}cpu {{.MemTotal}}'
sudo systemctl status docker --no-pager -l | sed -n '1,6p'

(Ergebnis: bestätigt Docker-Version, ob daemon läuft, Basisressourcen.)
2) Smoke-Test — hello-world

Prüft, ob Images starten und Netzwerk klappt:

sudo docker run --rm hello-world

Wenn das sauber läuft -> Docker ist funktional. Wenn du danach kein sudo verwenden willst, melde dich ab/neu oder führ newgrp docker aus (siehe Punkt 4).
3) Aufräumen (optional, wenn alte Reste Probleme machen)

Diese Befehle entfernen gestoppte Container, nicht genutzte Images/Netzwerke — sicher, aber unwiderruflich für nicht-gepushte Images:

# Liste, dann prune wenn ok
docker ps -a
# Stopp & entferne alle gestoppten
docker container prune -f
# Ungenutzte images/volumes/networks
docker image prune -af
docker network prune -f
docker volume prune -f
# Oder ein radikaler System-Cleanup
docker system prune -af --volumes

Die „stale sandbox / endpoint key modified“ Fehler kommen oft von alten Netzwerkendpunkten — docker network prune hilft meist.
4) Zugriff ohne sudo (falls noch nicht gemacht)

Wenn du docker ohne sudo willst:

sudo usermod -aG docker "$USER"

#!/usr/bin/env bash
set -euo pipefail
# tools/setup_python_env.sh
# Erstellt eine lokale Python-Umgebung via pyenv (falls nicht installiert) und eine venv.

PROJECT_ROOT="$(pwd)"
VENV_NAME="venv313"
PYTHON_VERSION="${1:-3.11.4}"   # optionales Argument: gewünschte Python-Version

echo "Project root: $PROJECT_ROOT"
echo "Gewünschte Python-Version: $PYTHON_VERSION"

# 1) Prüfe aktuelles Python
echo "-- aktuelles system python --"
python3 --version || true

# 2) Installiere pyenv falls nicht vorhanden
if ! command -v pyenv >/dev/null 2>&1; then
  echo "pyenv nicht gefunden -> installiere pyenv (dies kann ein paar Minuten dauern)"
  curl https://pyenv.run | bash
  export PATH="$HOME/.pyenv/bin:$PATH"
  eval "$(pyenv init -)"
  eval "$(pyenv virtualenv-init -)"
else
  echo "pyenv vorhanden"
fi

# 3) Stelle sicher, dass pyenv in dieser Session verfügbar ist
export PATH="$HOME/.pyenv/bin:$PATH"
eval "$(pyenv init -)"
eval "$(pyenv virtualenv-init -)"

# 4) Installiere gewünschte Python-Version, falls noch nicht vorhanden
if ! pyenv versions --bare | grep -q "^${PYTHON_VERSION}\$"; then
  echo "Installiere Python $PYTHON_VERSION via pyenv..."
  pyenv install "$PYTHON_VERSION"
else
  echo "Python $PYTHON_VERSION bereits installiert"
fi

# 5) Setze lokale Python-Version für dieses Verzeichnis
pyenv local "$PYTHON_VERSION"

# 6) Erstelle venv (im Projekt-Root)
if [ -d "${VENV_NAME}" ]; then
  echo "Venv ${VENV_NAME} existiert bereits -> benutze vorhandene Umgebung"
else
  echo "Erstelle venv ${VENV_NAME} mit python $(python --version)"
  python -m venv "${VENV_NAME}"
fi

# 7) Aktiviere venv für diese Session
# shellcheck disable=SC1090
source "${VENV_NAME}/bin/activate"
echo "Venv aktiviert: $(which python) $(python --version)"

# 8) Upgrade pip + install requirements
python -m pip install --upgrade pip setuptools wheel
if [ -f "requirements.txt" ]; then
  pip install -r requirements.txt
else
  echo "requirements.txt nicht gefunden — überspringe pip install"
fi

# 9) Optional: Tests ausführen
if command -v pytest >/dev/null 2>&1; then
  echo "Führe pytest aus..."
  pytest -q || echo "Tests: einige Tests fehlgeschlagen (siehe pytest output)"
else
  echo "pytest nicht installiert, überspringe Tests"
fi

echo "Setup abgeschlossen. Aktivieren: source ${VENV_NAME}/bin/activate"
echo "Starte den Agent-Server (Beispiel): cd dashboard/backend && python3 agent_server.py"

#!/usr/bin/env python3
# dashboard/backend/agent_server.py
# Minimal, self-contained Agent Server for local development.
# - FastAPI app exposing the basic endpoints used by the dashboard and CLI.
# - In-memory AgentManager with optional simple JSON persistence to ./data/agent_registry.json
# - Single-file drop-in when dashboard/backend is empty.
#
# Usage:
#   uvicorn agent_server:app --host 127.0.0.1 --port 12344 --reload
# or run directly:
#   python3 agent_server.py
#
# Port recommendation: default 12344 (do not use 8080).

from __future__ import annotations
import json
import uuid
import threading
from datetime import datetime
from pathlib import Path
from typing import Dict, List, Optional, Any

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

# ---- Config ----
DEFAULT_PORT = int(__import__("os").environ.get("PORT", "12344"))
DATA_DIR = Path(__file__).parent / "data"
REGISTRY_FILE = DATA_DIR / "agent_registry.json"
ALLOWED_PORTS = list(range(12344, 12400))
FORBIDDEN_PORTS = [8080]

# ---- Ensure data dir ----
DATA_DIR.mkdir(parents=True, exist_ok=True)


# ---- Models ----
class CreateAgentRequest(BaseModel):
    name: str = Field(..., min_length=1)
    type: Optional[str] = "prompt_processor"


class PromptRequest(BaseModel):
    prompt: str = Field(..., min_length=1)
    agent_id: Optional[str] = None
    context: Optional[Dict[str, Any]] = None


class GenericResponse(BaseModel):
    success: bool
    message: Optional[str] = None
    data: Optional[Any] = None


# ---- Simple Agent & Manager ----
class SimpleAgent:
    def __init__(self, name: str, agent_type: str = "prompt_processor"):
        self.id = str(uuid.uuid4())
        self.name = name
        self.agent_type = agent_type
        self.created_at = datetime.utcnow().isoformat()
        self.last_activity = self.created_at
        self.state = "active"
        self.memory: Dict[str, Any] = {}
        self.routes: List[str] = []

    def get_info(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "type": self.agent_type,
            "created_at": self.created_at,
            "last_activity": self.last_activity,
            "state": self.state,
            "memory_size": len(self.memory),
            "routes": self.routes,
        }

    def process_prompt(self, prompt: str) -> str:
        self.last_activity = datetime.utcnow().isoformat()
        # Minimal, deterministic fake processing (no external LLM calls).
        p = prompt.strip()
        if not p:
            return "Empty prompt."
        lower = p.lower()
        if "hallo" in lower or "hello" in lower:
            return f"Hallo — ich bin {self.name}. Wie kann ich helfen?"
        if "status" in lower:
            return f"Agent {self.name} ist {self.state}. Speicher: {len(self.memory)} Einträge."
        if "hilfe" in lower or "help" in lower:
            return (
                "Verfügbare Befehle: 'Agent erstellen', 'Status', 'Hilfe', oder beliebige Prompt-Eingabe."
            )
        # generic echo with timestamp
        ts = datetime.utcnow().strftime("%Y-%m-%d %H:%M:%SZ")
        # store to memory
        self.memory[f"prompt_{ts}"] = {"prompt": p, "ts": ts}
        return f"Antwort ({ts}): {p}"


class AgentManager:
    def __init__(self, persist_file: Optional[Path] = REGISTRY_FILE):
        self._lock = threading.RLock()
        self.agents: Dict[str, SimpleAgent] = {}
        self.persist_file = persist_file
        self._load_registry()
        if not any(a.agent_type == "prompt_processor" for a in self.agents.values()):
            # create default main agent
            a = SimpleAgent("MainPromptAgent", "prompt_processor")
            a.routes = ["/prompt", "/dashboard"]
            self.register_agent(a)

    def register_agent(self, agent: SimpleAgent) -> str:
        with self._lock:
            self.agents[agent.id] = agent
            self._save_registry()
            return agent.id

    def create_agent(self, name: str, agent_type: str = "prompt_processor") -> str:
        agent = SimpleAgent(name, agent_type)
        return self.register_agent(agent)

    def list_agents(self) -> List[Dict[str, Any]]:
        with self._lock:
            return [a.get_info() for a in self.agents.values()]

    def get_agent(self, agent_id: str) -> Optional[SimpleAgent]:
        return self.agents.get(agent_id)

    def get_agent_by_name(self, name: str) -> Optional[SimpleAgent]:
        for a in self.agents.values():
            if a.name == name:
                return a
        return None

    def process_prompt(self, prompt: str, agent_id: Optional[str] = None) -> Dict[str, Any]:
        with self._lock:
            agent = self.get_agent(agent_id) if agent_id else self.get_agent_by_name("MainPromptAgent")
            if not agent:
                return {"success": False, "message": "No suitable agent found"}
            response = agent.process_prompt(prompt)
            self._save_registry()
            return {"success": True, "response": response, "agent_id": agent.id, "agent_name": agent.name}

    def get_system_stats(self) -> Dict[str, Any]:
        with self._lock:
            stats = {
                "total_agents": len(self.agents),
                "active_agents": len([a for a in self.agents.values() if a.state == "active"]),
                "total_routes": sum(len(a.routes) for a in self.agents.values()),
            }
            return stats

    def self_regenerate(self) -> Dict[str, Any]:
        with self._lock:
            log: List[str] = []
            # Reactivate inactive agents
            for a in self.agents.values():
                if a.state != "active":
                    a.state = "active"
                    log.append(f"Reactivated {a.name}")
            # Ensure prompt agent exists
            if not any(a.agent_type == "prompt_processor" for a in self.agents.values()):
                new_id = self.create_agent("RegeneratedPromptAgent")
                log.append(f"Created prompt agent {new_id}")
            self._save_registry()
            log.append("Saved registry")
            return {"success": True, "regeneration_log": log}

    # Persistence (very small, atomic write)
    def _save_registry(self):
        if not self.persist_file:
            return
        tmp = self.persist_file.with_suffix(".tmp")
        payload = {
            "agents": {aid: self._serialize_agent(a) for aid, a in self.agents.items()},
            "last_saved": datetime.utcnow().isoformat(),
        }
        tmp.parent.mkdir(parents=True, exist_ok=True)
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(payload, f, indent=2, ensure_ascii=False)
        tmp.replace(self.persist_file)

    def _load_registry(self):
        if not self.persist_file or not self.persist_file.exists():
            return
        try:
            with self.persist_file.open("r", encoding="utf-8") as f:
                payload = json.load(f)
            agents = payload.get("agents", {})
            for aid, ad in agents.items():
                a = SimpleAgent(ad.get("name", f"agent-{aid}"), ad.get("type", "generic"))
                a.id = aid
                a.created_at = ad.get("created_at", a.created_at)
                a.last_activity = ad.get("last_activity", a.last_activity)
                a.state = ad.get("state", a.state)
                a.memory = ad.get("memory", {})
                a.routes = ad.get("routes", [])
                self.agents[aid] = a
        except Exception:
            # ignore load errors, start fresh
            return

    @staticmethod
    def _serialize_agent(a: SimpleAgent) -> Dict[str, Any]:
        return {
            "id": a.id,
            "name": a.name,
            "type": a.agent_type,
            "created_at": a.created_at,
            "last_activity": a.last_activity,
            "state": a.state,
            "memory": a.memory,
            "routes": a.routes,
        }


# ---- App ----
app = FastAPI(title="Agent Server (minimal)", version="0.1")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["GET", "POST", "OPTIONS"],
    allow_headers=["*"],
)

manager = AgentManager()


@app.get("/agent/health", response_model=Dict[str, Any])
def health():
    return {
        "status": "healthy",
        "timestamp": datetime.utcnow().isoformat(),
        "allowed_ports": ALLOWED_PORTS,
        "forbidden_ports": FORBIDDEN_PORTS,
    }


@app.get("/agent/list", response_model=List[Dict[str, Any]])
def list_agents():
    return manager.list_agents()


@app.get("/agent/stats", response_model=Dict[str, Any])
def stats():
    return manager.get_system_stats()


@app.get("/agent/routes", response_model=Dict[str, Any])
def routes():
    with manager._lock:
        return {
            aid: {"agent_name": a.name, "routes": a.routes}
            for aid, a in manager.agents.items()
        }


@app.post("/agent/create", response_model=GenericResponse)
def create_agent(req: CreateAgentRequest):
    try:
        aid = manager.create_agent(req.name, req.type)
        return GenericResponse(success=True, message="Agent created", data={"agent_id": aid})
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/agent/prompt", response_model=Dict[str, Any])
def prompt(req: PromptRequest):
    if not req.prompt:
        raise HTTPException(status_code=400, detail="prompt required")
    res = manager.process_prompt(req.prompt, req.agent_id)
    return res


@app.post("/agent/regenerate", response_model=GenericResponse)
def regenerate():
    try:
        res = manager.self_regenerate()
        return GenericResponse(success=True, message="Regenerated", data=res)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


# small convenience endpoints for conversation retrieval
@app.get("/agent/{agent_id}/conversations", response_model=List[Dict[str, Any]])
def get_conversations(agent_id: str, limit: int = 10):
    a = manager.get_agent(agent_id)
    if not a:
        raise HTTPException(status_code=404, detail="agent not found")
    convs = []
    # memory keys are stored as prompt_{ts}; return latest entries
    for k, v in list(a.memory.items())[-limit:]:
        convs.append({"key": k, "entry": v})
    return convs

## 💪 Robuste Version von `start_dev_server.sh`

```bash
#!/bin/bash

# tools/start_dev_server.sh
# Starte lokalen Agent-Server mit FastAPI

set -e

# Pfade
VENV_DIR="./venv313"
APP_DIR="dashboard/backend"
APP_MODULE="agent_server:app"
HOST="127.0.0.1"
PORT="12344"

# Farben
RED='\033[0;31m'
GREEN='\033[0;32m'
NC='\033[0m' # No Color

echo ""
echo -e "${GREEN}🌱 Starte lokalen Agent-Server...${NC}"

# Prüfe, ob venv existiert
if [ ! -d "$VENV_DIR" ]; then
    echo -e "${RED}❌ Virtuelle Umgebung nicht gefunden unter: ${VENV_DIR}${NC}"
    echo "Bitte ausführen:"
    echo "  python3 -m venv venv313"
    echo "  source venv313/bin/activate"
    echo "  pip install -r requirements.txt"
    exit 1
fi

# Aktiviere venv
source "$VENV_DIR/bin/activate"

# Prüfe, ob uvicorn verfügbar
if ! command -v uvicorn &> /dev/null; then
    echo -e "${RED}❌ 'uvicorn' ist nicht installiert.${NC}"
    echo "Bitte ausführen: pip install -r requirements.txt"
    exit 1
fi

# Prüfe App-Verzeichnis
if [ ! -d "$APP_DIR" ]; then
    echo -e "${RED}❌ App-Verzeichnis nicht gefunden: ${APP_DIR}${NC}"
    echo "Pfad prüfen. Erwartet Projektstruktur wie:"
    echo "  $APP_DIR/$APP_MODULE"
    exit 1
fi

# Starte Server
echo -e "${GREEN}✅ Alles bereit. Server wird gestartet unter http://$HOST:$PORT${NC}"
echo ""

uvicorn --app-dir "$APP_DIR" "$APP_MODULE" --host "$HOST" --port "$PORT" --reload
```
